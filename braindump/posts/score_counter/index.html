A round-based board game score counter PWA, inspired by the Android app “Score Counter” by Torsten Hoffmann (de.th.scorecounter). Uses a spreadsheet-style table with players as columns and rounds as rows.

Features: add/remove players (up to 8), round-by-round score entry via a bottom bar, tap any cell to edit or delete a round, sticky headers and totals row with winner highlighting, scoreboard ranking modal, save/load games, export to CSV, new game prompts for game name (prefilled from previous) and preserves player names, wake lock to keep screen on, websocket sync indicator dot.

Stack: Yjs + y-indexeddb + y-websocket (esm.sh with pinned deps), vanilla JS with imperative DOM rendering, vanilla CSS (dark theme), service worker (network-first for HTML, cache-first for CDN).

// ── Yjs setup (shared with tally_score_tracker) ──
var ydoc = new Y.Doc();
var persistence = new IndexeddbPersistence('tally', ydoc);
var yPlayers = ydoc.getArray('currentPlayers'); // Y.Map {name}
var yRounds  = ydoc.getArray('rounds');         // [[d0, d1, ...], ...]
var yGames   = ydoc.getArray('savedGames');     // [{id,date,gameName,players,rounds}]
var yRoster  = ydoc.getArray('sc_roster');      // [{name:'...'}] persistent player roster
var yMeta    = ydoc.getMap('meta');              // gameName
var COLORS   = ['#e94560','#4ecca3','#533483','#f9a826','#3fc1c9','#fc5185','#8dc6ff','#6c5ce7'];
var MAX_PLAYERS = 8;

window.ydoc = ydoc;
window.Y = Y;

// ── DOM refs ──
var $ = function(id){ return document.getElementById(id); };

// ── Helpers ──
function esc(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/"/g,'&quot;'); }

function getPlayers(){
  var a = [];
  for(var i=0;i<yPlayers.length;i++) a.push({name: yPlayers.get(i).get('name')});
  return a;
}
function getRounds(){
  var a = [];
  for(var i=0;i<yRounds.length;i++) a.push(yRounds.get(i));
  return a;
}
function getSavedGames(){
  var a = [];
  for(var i=0;i<yGames.length;i++) a.push(yGames.get(i));
  return a;
}

function totals(){
  var players = getPlayers();
  var rounds = getRounds();
  var t = players.map(function(){ return 0; });
  rounds.forEach(function(r){
    r.forEach(function(v,i){ t[i] += (v||0); });
  });
  return t;
}

// ── Render ──
function render(){
  var players = getPlayers();
  var rounds = getRounds();
  var np = players.length;
  var hasPlayers = np > 0;

  $('emptyState').style.display = hasPlayers ? 'none' : '';
  $('scoreTable').style.display = hasPlayers ? '' : 'none';

  var gn = yMeta.get('gameName') || '';
  if(document.activeElement !== $('gameName')) $('gameName').value = gn;

  // Header row
  var hdr = '<th>#</th>';
  players.forEach(function(p, i){
    hdr += '<th style="border-bottom:3px solid '+COLORS[i%8]+'">' +
      '<input class="player-name-input" value="'+esc(p.name)+'" data-pidx="'+i+'" placeholder="Player '+(i+1)+'" spellcheck="false">' +
      '<button class="remove-player" data-pidx="'+i+'" title="Remove">&#x2715; remove</button></th>';
  });
  $('headerRow').innerHTML = hdr;

  // Body rows
  var body = '';
  rounds.forEach(function(r, ri){
    body += '<tr><td>'+(ri+1)+'</td>';
    r.forEach(function(v, pi){
      var cls = 'score-cell' + (v>0?' positive':'') + (v<0?' negative':'');
      body += '<td class="'+cls+'" data-round="'+ri+'" data-player="'+pi+'">'+v+'</td>';
    });
    body += '</tr>';
  });

  // Input row (always last before totals)
  body += '<tr class="entry-row"><td><button class="entry-confirm" id="entryConfirm" title="Add round">+</button></td>';
  players.forEach(function(p, i){
    body += '<td><input type="number" readonly class="entry-input" data-pidx="'+i+'" id="entry_'+i+'" placeholder="0" data-pname="'+esc(p.name)+'"></td>';
  });
  body += '</tr>';
  $('scoreBody').innerHTML = body;

  // Totals row
  var t = totals();
  var maxT = Math.max.apply(null, t.length ? t : [0]);
  var foot = '<td>&#x03a3;</td>';

$('numpad').onclick = function(e) {
  var btn = e.target.closest('.np-btn');
  if (!btn) return;
  var key = btn.dataset.key;
  var delta = btn.dataset.delta;

  if (delta) {
    // Quick increment
    var cur = parseInt(npTarget && npTarget.value) || 0;
    cur += parseInt(delta);
    npNeg = cur < 0;
    npRaw = Math.abs(cur).toString();
    npSync();
    return;
  }

  if (key >= '0' && key <= '9') {
    npRaw += key;
    npSync();
  } else if (key === '00') {
    npRaw += '00';
    npSync();
  } else if (key === 'del') {
    npRaw = npRaw.slice(0, -1);
    npSync();
  } else if (key === 'clear') {
    npRaw = '';
    npNeg = false;
    npSync();
  } else if (key === 'sign') {
    npNeg = !npNeg;
    npSync();
  } else if (key === 'next') {
    if (!npTarget) return;
    var idx = parseInt(npTarget.dataset.pidx);
    var next = $('entry_' + (idx + 1));
    if (next) openNumpad(next);
    else { addRound(); }
  } else if (key === 'done') {
    closeNumpad();
  }

async function handleQRResult(raw) {
  stopScan();
  var status = $('scanStatus');
  status.classList.add('open');
  status.textContent = 'Exchanging code...';

  try {
    var data = JSON.parse(raw);
    if (!data.code || !data.exchange) throw new Error('Invalid QR data');

    var exchangeUrl = location.protocol === 'http:'
        ? location.origin + '/authorizationserver/exchange'
        : data.exchange;

    var resp = await fetch(exchangeUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({ code: data.code }),
    });

    if (!resp.ok) {
      var err = await resp.json().catch(function() { return {}; });
      throw new Error(err.error || 'Exchange failed');
    }

    var wsUrl = location.protocol === 'http:'
        ? 'ws://' + location.host + '/ywebsocket'
        : 'wss://konubinix.eu/ywebsocket';
    localStorage.setItem('ywebsocket_url', wsUrl);
    status.textContent = 'Authorized! Reloading...';

  } catch (e) {
    status.textContent = 'Error: ' + e.message;
    setTimeout(function() { status.classList.remove('open'); }, 3000);
  }
}

Application logic

Imports and CRDT setup

The Yjs document (ydoc) is the single source of truth, shared with the card-based Tally app (tally_score_tracker.org). Both apps read and write the same shared types:

  • currentPlayers (Y.Array of Y.Maps): each Y.Map has name
  • rounds (Y.Array of arrays): each inner array holds one delta per player
  • savedGames (Y.Array): archive of finished games
  • meta (Y.Map): game-level metadata (currently just gameName)
  • sc_roster (Y.Array of plain objects): persistent player roster (score counter only)

Scores are derived by summing rounds per player column. No score or history fields on players — concurrent round appends merge correctly via Y.Array.

IndexeddbPersistence keeps a local copy in the browser’s IndexedDB so the app works fully offline. Alpine is imported but not used for rendering — it’s kept in the import map for potential future use.

DOM refs and helpers

Short $() helper for getElementById, HTML escaping, and three read helpers that snapshot Yjs arrays into plain JS arrays. totals() computes the column-wise sum of all rounds.

// ── DOM refs ──
var $ = function(id){ return document.getElementById(id); };

// ── Helpers ──
function esc(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/"/g,'&quot;'); }

function getPlayers(){
  var a = [];
  for(var i=0;i<yPlayers.length;i++) a.push({name: yPlayers.get(i).get('name')});
  return a;
}
function getRounds(){
  var a = [];
  for(var i=0;i<yRounds.length;i++) a.push(yRounds.get(i));
  return a;
}
function getSavedGames(){
  var a = [];
  for(var i=0;i<yGames.length;i++) a.push(yGames.get(i));
  return a;
}

function totals(){
  var players = getPlayers();
  var rounds = getRounds();
  var t = players.map(function(){ return 0; });
  rounds.forEach(function(r){
    r.forEach(function(v,i){ t[i] += (v||0); });
  });
  return t;
}

Render function

Imperative DOM rendering that reads from Yjs and rebuilds the table header (player names with color-coded bottom borders), body rows (one per round, each cell clickable), totals footer (winner highlighted in gold), and the entry bar (one number input per player with a confirm button).

function render(){
  var players = getPlayers();
  var rounds = getRounds();
  var np = players.length;
  var hasPlayers = np > 0;

  $('emptyState').style.display = hasPlayers ? 'none' : '';
  $('scoreTable').style.display = hasPlayers ? '' : 'none';

      '<button class="remove-player" data-pidx="'+i+'" title="Remove">&#x2715; remove</button></th>';
  });
  $('headerRow').innerHTML = hdr;

  // Body rows
  var body = '';
  rounds.forEach(function(r, ri){
    body += '<tr><td>'+(ri+1)+'</td>';
    r.forEach(function(v, pi){
      var cls = 'score-cell' + (v>0?' positive':'') + (v<0?' negative':'');
      body += '<td class="'+cls+'" data-round="'+ri+'" data-player="'+pi+'">'+v+'</td>';
    });
    body += '</tr>';
  });

  // Input row (always last before totals)
  body += '<tr class="entry-row"><td><button class="entry-confirm" id="entryConfirm" title="Add round">+</button></td>';
  players.forEach(function(p, i){
    body += '<td><input type="number" readonly class="entry-input" data-pidx="'+i+'" id="entry_'+i+'" placeholder="0" data-pname="'+esc(p.name)+'"></td>';
  });
  body += '</tr>';

Player management

Removing a player deletes the Y.Map entry from yPlayers and splices the corresponding index out of every round (deleting empty rounds).

// ── Player management ──
function removePlayer(idx){
  var pm = yPlayers.get(idx);
  if(!confirm('Remove ' + pm.get('name') + '?')) return;
  yPlayers.delete(idx, 1);
  for(var i=0;i<yRounds.length;i++){
    var r = yRounds.get(i).slice();
    r.splice(idx, 1);
    yRounds.delete(i, 1);
    if(r.length > 0) yRounds.insert(i, [r]);
    else { i--; }
  }
  render();
}

  $('npValue').textContent = val || '0';
  if (npTarget) npTarget.value = val || '';
}

function openNumpad(input) {
  npTarget = input;
  var cur = parseInt(input.value) || 0;
  npNeg = cur < 0;
  npRaw = Math.abs(cur).toString();
  if (cur === 0) npRaw = '';
  $('npLabel').textContent = input.dataset.pname || '';
  npSync();
  document.querySelectorAll('.entry-input').forEach(function(el) { el.classList.remove('active-pad'); });
  input.classList.add('active-pad');
  $('numpad').classList.add('open');
}

function closeNumpad() {
  $('numpad').classList.remove('open');
  if (npTarget) npTarget.classList.remove('active-pad');
  npTarget = null;
}

$('numpad').onclick = function(e) {
  var btn = e.target.closest('.np-btn');
  if (!btn) return;
  var key = btn.dataset.key;
  var delta = btn.dataset.delta;

  if (delta) {
    // Quick increment
    var cur = parseInt(npTarget && npTarget.value) || 0;
    cur += parseInt(delta);
    npNeg = cur < 0;
    npRaw = Math.abs(cur).toString();
    npSync();
    return;
  }

  if (key >= '0' && key <= '9') {
    npRaw += key;
    npSync();
  } else if (key === '00') {
    npRaw += '00';
    npSync();
  } else if (key === 'del') {
    npRaw = npRaw.slice(0, -1);
    npSync();
  } else if (key === 'clear') {
    npRaw = '';
    npNeg = false;
    npSync();
  } else if (key === 'sign') {
    npNeg = !npNeg;
    npSync();
  } else if (key === 'next') {
    if (!npTarget) return;
    var idx = parseInt(npTarget.dataset.pidx);
    var next = $('entry_' + (idx + 1));
    if (next) openNumpad(next);
    else { addRound(); }
  } else if (key === 'done') {
    closeNumpad();
  }
};

Tapping the scan button opens html5-qrcode which handles camera selection, preview and QR decoding. On success the authorization code is exchanged for a cookie.

// ── QR Scan ──
var html5QrScanner = null;

function stopScan() {
  if (html5QrScanner) {
    html5QrScanner.clear();
    html5QrScanner = null;
  }
  $('scanOverlay').classList.remove('open');

async function handleQRResult(raw) {
  stopScan();
  var status = $('scanStatus');
  status.classList.add('open');
  status.textContent = 'Exchanging code...';

  try {
    var data = JSON.parse(raw);
    if (!data.code || !data.exchange) throw new Error('Invalid QR data');

    var exchangeUrl = location.protocol === 'http:'
        ? location.origin + '/authorizationserver/exchange'
        : data.exchange;

    var resp = await fetch(exchangeUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({ code: data.code }),
    });

    if (!resp.ok) {
      var err = await resp.json().catch(function() { return {}; });
      throw new Error(err.error || 'Exchange failed');
    }

    var wsUrl = location.protocol === 'http:'
        ? 'ws://' + location.host + '/ywebsocket'
        : 'wss://konubinix.eu/ywebsocket';
    localStorage.setItem('ywebsocket_url', wsUrl);
    status.textContent = 'Authorized! Reloading...';

    setTimeout(function() { location.reload(); }, 1000);
  } catch (e) {
    status.textContent = 'Error: ' + e.message;
    setTimeout(function() { status.classList.remove('open'); }, 3000);
  }
}

  );
};
