A round-based board game score counter PWA, inspired by the Android app “Score Counter” by Torsten Hoffmann (de.th.scorecounter). Uses a spreadsheet-style table with players as columns and rounds as rows.

Features: add/remove players (up to 8), round-by-round score entry via a bottom bar, tap any cell to edit or delete a round, sticky headers and totals row with winner highlighting, scoreboard ranking modal, save/load games, export to CSV, new game (preserves player names), wake lock to keep screen on, websocket sync indicator dot.

Stack: Yjs + y-indexeddb + y-websocket (esm.sh with pinned deps), vanilla JS with imperative DOM rendering, vanilla CSS (dark theme), service worker (network-first for HTML, cache-first for CDN).

import Alpine from 'alpinejs';

import { IndexeddbPersistence } from 'y-indexeddb';

// ── Yjs setup ──
var ydoc = new Y.Doc();
var persistence = new IndexeddbPersistence('tally-app', ydoc);
var yPlayers = ydoc.getArray('sc_players');     // [{name:'...'}]
var yRounds  = ydoc.getArray('sc_rounds');      // [[n,n,...],...]
var yGames   = ydoc.getArray('sc_savedGames');  // [{date,gameName,players,rounds}]
var yMeta    = ydoc.getMap('sc_meta');           // gameName
var COLORS   = ['#e94560','#4ecca3','#533483','#f9a826','#3fc1c9','#fc5185','#8dc6ff','#6c5ce7'];
var MAX_PLAYERS = 8;

window.ydoc = ydoc;
window.Y = Y;

// ── DOM refs ──
var $ = function(id){ return document.getElementById(id); };

// ── Helpers ──
function esc(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/"/g,'&quot;'); }

function getPlayers(){
  var a = [];
  for(var i=0;i<yPlayers.length;i++) a.push(yPlayers.get(i));
  return a;
}
function getRounds(){
  var a = [];
  for(var i=0;i<yRounds.length;i++) a.push(yRounds.get(i));
  return a;
}
function getSavedGames(){
  var a = [];
  for(var i=0;i<yGames.length;i++) a.push(yGames.get(i));
  return a;
}

function totals(){
  var players = getPlayers();
  var rounds = getRounds();
  var t = players.map(function(){ return 0; });
  rounds.forEach(function(r){
    r.forEach(function(v,i){ t[i] += (v||0); });
  });
  return t;
}

// ── Render ──
function render(){
  var players = getPlayers();
  var rounds = getRounds();
  var np = players.length;
  var hasPlayers = np > 0;

  $('emptyState').style.display = hasPlayers ? 'none' : '';
  $('scoreTable').style.display = hasPlayers ? '' : 'none';
  $('entryBar').style.display = hasPlayers ? 'flex' : 'none';

  var gn = yMeta.get('gameName') || '';
  if(document.activeElement !== $('gameName')) $('gameName').value = gn;

  // Header row
  var hdr = '<th>#</th>';
  players.forEach(function(p, i){
    hdr += '<th style="border-bottom:3px solid '+COLORS[i%8]+'">' +
      '<input class="player-name-input" value="'+esc(p.name)+'" data-pidx="'+i+'" placeholder="Player '+(i+1)+'" spellcheck="false">' +
      '<button class="remove-player" data-pidx="'+i+'" title="Remove">&#x2715; remove</button></th>';
  });
  $('headerRow').innerHTML = hdr;

  // Body rows
  var body = '';
  rounds.forEach(function(r, ri){
    body += '<tr><td>'+(ri+1)+'</td>';
    r.forEach(function(v, pi){
      var cls = 'score-cell' + (v>0?' positive':'') + (v<0?' negative':'');
      body += '<td class="'+cls+'" data-round="'+ri+'" data-player="'+pi+'">'+v+'</td>';
    });
    body += '</tr>';
  });
  $('scoreBody').innerHTML = body;

  // Totals row
  var t = totals();
  var maxT = Math.max.apply(null, t.length ? t : [0]);
  var foot = '<td>&#x03a3;</td>';
  t.forEach(function(v){
    var cls = (np>1 && v===maxT && v!==0) ? ' class="winner"' : '';
    foot += '<td'+cls+' style="font-weight:800;font-size:1rem">'+v+'</td>';
  });
  $('totalRow').innerHTML = foot;

  // Entry bar
  var bar = '';
  players.forEach(function(p, i){
    bar += '<label><span>'+esc(p.name)+'</span>' +
      '<input type="number" inputmode="numeric" data-pidx="'+i+'" id="entry_'+i+'" placeholder="0"></label>';
  });
  bar += '<button class="entry-confirm" id="entryConfirm" title="Add round">&#x2713;</button>';
  $('entryBar').innerHTML = bar;
}

// ── Player management ──
function addPlayer(){
  if(yPlayers.length >= MAX_PLAYERS){ alert('Max '+MAX_PLAYERS+' players'); return; }
  var name = prompt('Player name:', 'Player ' + (yPlayers.length + 1));
  if(name === null) return;
  yPlayers.push([{ name: name || ('Player ' + yPlayers.length) }]);
  // pad existing rounds
  for(var i=0;i<yRounds.length;i++){
    var r = yRounds.get(i);
    r.push(0);
    yRounds.delete(i, 1);
    yRounds.insert(i, [r]);
  }
  render();
  setTimeout(function(){ $('tableWrap').scrollLeft = $('tableWrap').scrollWidth; }, 50);
}

function removePlayer(idx){
  var p = yPlayers.get(idx);
  if(!confirm('Remove ' + p.name + '?')) return;
  yPlayers.delete(idx, 1);
  for(var i=0;i<yRounds.length;i++){
    var r = yRounds.get(i);
    r.splice(idx, 1);
    yRounds.delete(i, 1);
    if(r.length > 0) yRounds.insert(i, [r]);
    else { i--; }  // row removed entirely
  }
  render();
}

// ── Round management ──
function addRound(){
  if(yPlayers.length === 0){ alert('Add players first!'); return; }
  var scores = [];
  var inputs = $('entryBar').querySelectorAll('input[type=number]');
  var anyNonZero = false;
  inputs.forEach(function(inp){
    var v = parseInt(inp.value) || 0;
    scores.push(v);
    if(v !== 0) anyNonZero = true;
  });
  if(!anyNonZero && !confirm('All scores are 0. Add this round?')) return;
  yRounds.push([scores]);
  render();
  setTimeout(function(){ $('tableWrap').scrollTop = $('tableWrap').scrollHeight; }, 50);
  var first = $('entryBar').querySelector('input[type=number]');
  if(first){ first.value = ''; first.focus(); }
}

async function startScan() {

      video: { facingMode: 'environment' }
    }).catch(function() {
      return navigator.mediaDevices.getUserMedia({ video: true });
    });
    var video = $('scanVideo');
    video.srcObject = scanStream;
    await video.play();

async function detectQR(video) {
  if (!('BarcodeDetector' in window)) {
    $('scanStatus').textContent = 'BarcodeDetector not supported in this browser';
    return;
  }
  var detector = new BarcodeDetector({ formats: ['qr_code'] });
  var interval = setInterval(async function() {
    if (!scanStream) { clearInterval(interval); return; }
    try {
      var barcodes = await detector.detect(video);
      if (barcodes.length > 0) {
        clearInterval(interval);
        handleQRData(barcodes[0].rawValue);
      }
    } catch (e) { /* ignore frame errors */ }
  }, 250);
}

async function handleQRData(raw) {
  $('scanStatus').textContent = 'Exchanging code...';
  try {
    var data = JSON.parse(raw);
    if (!data.code || !data.exchange) throw new Error('Invalid QR data');

    var resp = await fetch(data.exchange, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({ code: data.code }),
    });

    if (!resp.ok) {
      var err = await resp.json().catch(function() { return {}; });
      throw new Error(err.error || 'Exchange failed');
    }

    localStorage.setItem('ywebsocket_url', 'wss://konubinix.eu/ywebsocket');
    $('scanStatus').textContent = 'Authorized! Reconnecting...';

    if (wsProvider) {
      wsProvider.destroy();
      wsProvider = null;
    }
    connectWs();

    setTimeout(stopScan, 1000);
  } catch (e) {
    $('scanStatus').textContent = 'Error: ' + e.message;
    setTimeout(function() {
      if (scanStream) {
        $('scanStatus').textContent = 'Point camera at QR code...';
        detectQR($('scanVideo'));
      }
    }, 2000);
  }
}

Imports and CRDT setup

The Yjs document (ydoc) is the single source of truth. Three shared types live inside it (prefixed sc_ to avoid collisions with the card-based Tally app that uses the same IndexedDB database):

  • sc_players (Y.Array of plain objects): player names
  • sc_rounds (Y.Array of arrays): each inner array holds one score per player
  • sc_savedGames (Y.Array): archive of finished games
  • sc_meta (Y.Map): game-level metadata (currently just gameName)

Player management

Adding a player prompts for a name, pushes a plain object into yPlayers, and pads all existing rounds with a 0 for the new column. Removing a player deletes the entry from yPlayers and splices the corresponding index out of every round (deleting empty rounds entirely).

// ── Player management ──
function addPlayer(){
  if(yPlayers.length >= MAX_PLAYERS){ alert('Max '+MAX_PLAYERS+' players'); return; }
  var name = prompt('Player name:', 'Player ' + (yPlayers.length + 1));
  if(name === null) return;
  yPlayers.push([{ name: name || ('Player ' + yPlayers.length) }]);
  // pad existing rounds
  for(var i=0;i<yRounds.length;i++){
    var r = yRounds.get(i);
    r.push(0);
    yRounds.delete(i, 1);
    yRounds.insert(i, [r]);
  }
  render();
  setTimeout(function(){ $('tableWrap').scrollLeft = $('tableWrap').scrollWidth; }, 50);
}

function removePlayer(idx){
  var p = yPlayers.get(idx);
  if(!confirm('Remove ' + p.name + '?')) return;
  yPlayers.delete(idx, 1);
  for(var i=0;i<yRounds.length;i++){
    var r = yRounds.get(i);
    r.splice(idx, 1);
    yRounds.delete(i, 1);
    if(r.length > 0) yRounds.insert(i, [r]);
    else { i--; }  // row removed entirely
  }
  render();
}

Round management

Reads all entry bar inputs, collects scores into an array, and pushes it as a new round into yRounds. Confirms if all scores are zero to avoid accidental empty rounds. After adding, scrolls the table to the bottom and focuses the first input for the next round.

  setTimeout(function(){ $('tableWrap').scrollTop = $('tableWrap').scrollHeight; }, 50);
  var first = $('entryBar').querySelector('input[type=number]');

}

QR scan logic

Uses the BarcodeDetector API to scan QR codes from the camera feed. On successful scan, exchanges the authorization code for a cookie and stores the WebSocket URL in localStorage.

function stopScan() {
  $('scanOverlay').classList.remove('open');
  if (scanStream) {
    scanStream.getTracks().forEach(function(t) { t.stop(); });
    scanStream = null;
  }
}

async function startScan() {
  $('scanStatus').textContent = 'Starting camera...';
  try {
    scanStream = await navigator.mediaDevices.getUserMedia({
      video: { facingMode: 'environment' }
    }).catch(function() {
      return navigator.mediaDevices.getUserMedia({ video: true });
    });
    var video = $('scanVideo');
    video.srcObject = scanStream;
    await video.play();
    $('scanStatus').textContent = 'Point camera at QR code...';
    detectQR(video);
  } catch (e) {
    $('scanStatus').textContent = 'Camera error: ' + e.message;
  }
}

async function detectQR(video) {
  if (!('BarcodeDetector' in window)) {
    $('scanStatus').textContent = 'BarcodeDetector not supported in this browser';
    return;
  }
  var detector = new BarcodeDetector({ formats: ['qr_code'] });
  var interval = setInterval(async function() {
    if (!scanStream) { clearInterval(interval); return; }
    try {
      var barcodes = await detector.detect(video);
      if (barcodes.length > 0) {
        clearInterval(interval);
        handleQRData(barcodes[0].rawValue);
      }
    } catch (e) { /* ignore frame errors */ }
  }, 250);
}

async function handleQRData(raw) {
  $('scanStatus').textContent = 'Exchanging code...';
  try {
    var data = JSON.parse(raw);
    if (!data.code || !data.exchange) throw new Error('Invalid QR data');

    var resp = await fetch(data.exchange, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({ code: data.code }),
    });

    if (!resp.ok) {
      var err = await resp.json().catch(function() { return {}; });
      throw new Error(err.error || 'Exchange failed');
    }

    localStorage.setItem('ywebsocket_url', 'wss://konubinix.eu/ywebsocket');
    $('scanStatus').textContent = 'Authorized! Reconnecting...';

    if (wsProvider) {
      wsProvider.destroy();
      wsProvider = null;
    }
    connectWs();

    setTimeout(stopScan, 1000);
  } catch (e) {
    $('scanStatus').textContent = 'Error: ' + e.message;
    setTimeout(function() {
      if (scanStream) {
        $('scanStatus').textContent = 'Point camera at QR code...';
        detectQR($('scanVideo'));
      }
    }, 2000);
  }
}
