Giving AI agents partial access to my running qutebrowser

why
so those agents can help me investigate issues.
what
partial means I grant seeing and interacting apart.
rule
a solution that leaves how qutebrowser is launched unchanged is preferred.
  • scope: opening its DevTools port, which needs such a change, is the fallback.
how
the agent’s tools come from a small MCP server that sends commands to the socket a running qutebrowser listens on1, with seeing and interacting in separate tools.

def send(command, linger=0, anywhere=False):
    if AGENT_TAB and not anywhere:
        command = f":agent-run {command.removeprefix(':')}"
    name = hashlib.md5(getpass.getuser().encode()).hexdigest()
    path = runtime() / f"ipc-{name}"
    with socket.socket(socket.AF_UNIX) as ipc:

        ipc.sendall((json.dumps({"args": [command], "target_arg": None, "protocol_version": 1}) + "\n").encode())
        time.sleep(linger)

    deadline = time.monotonic() + wait

ACTING = {"calls": 0, "lock": threading.Lock()}


def title_acting(acting):
    try:
        act(call("titleActing", acting), CHECK_WAIT)
    except (RuntimeError, TimeoutError, OSError):
        pass


@contextlib.contextmanager
def titled():
    with ACTING["lock"]:
        ACTING["calls"] += 1
        if ACTING["calls"] == 1:
            title_acting(True)
    try:
        yield
    finally:
        with ACTING["lock"]:
            ACTING["calls"] -= 1
            if not ACTING["calls"]:
                title_acting(False)

      if (name == 'edit') trust.editing = true;
      if (name == 'earlier') earlier();
      if (name == 'wider') wider();
      if (name == 'narrower') narrower();
      if (name == 'later') later();
      if (name == 'keep') trust.editing = false;
      buttons();
    };
    trust.keyed = event => {
      const name = this.trustKeys[event.key];
      if (!trust.editing || !event.isTrusted || !name) return;
      event.stopImmediatePropagation();
      event.preventDefault();
      if (event.type == 'keydown') press(name);
    };
    for (const type of ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click'])
      window.addEventListener(type, trust.guard, true);
    for (const type of ['keydown', 'keypress', 'keyup'])
      window.addEventListener(type, trust.keyed, true);
    window.addEventListener('focusin', trust.pointed, true);
    const follow = () => {
      if (this.state.trust != trust) return;
      if (!part().isConnected) return this.untrust();
      const boxes = trust.parts().map(element => element.getBoundingClientRect()).filter(box => box.width || box.height);
      const left = Math.min(...boxes.map(box => box.left)), top = Math.min(...boxes.map(box => box.top));
      const box = {left, top, width: Math.max(...boxes.map(box => box.right)) - left,
                   height: Math.max(...boxes.map(box => box.bottom)) - top};
      const border = trust.editing ? 'dashed #800080' : 'dashed #008080';
      trust.frame.style.cssText = `position: fixed; z-index: 2147483645; pointer-events: none; border: 3px ${border};`
        + ` left: ${box.left - 3}px; top: ${box.top - 3}px; width: ${box.width}px; height: ${box.height}px`;
      trust.bar.style.cssText = 'position: fixed; z-index: 2147483647; background: #008080;'
        + ` left: ${Math.max(0, box.left - 3)}px; top: ${Math.max(0, box.top - 30)}px`;
      requestAnimationFrame(follow);
    };
    buttons();
    document.documentElement.append(trust.frame, trust.bar);
    follow();
  },

  untrust() {
    const trust = this.state.trust;
    if (!trust) return;
    this.state.trust = null;
    for (const element of trust.parts()) delete element.dataset.qutebrowserMcpTrusted;
    trust.frame.remove();
    trust.bar.remove();
    for (const type of ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click'])
      window.removeEventListener(type, trust.guard, true);
    for (const type of ['keydown', 'keypress', 'keyup'])
      window.removeEventListener(type, trust.keyed, true);
    window.removeEventListener('focusin', trust.pointed, true);
  },

  ask(element, question) {
    return new Promise(resolve => {
      const panel = document.createElement('div');
      const box = element.getBoundingClientRect();
      panel.dataset.qutebrowserMcpQuestion = '';
      panel.style.cssText = 'position: fixed; z-index: 2147483647; background: #800080; color: white;'
        + ` font: 14px sans-serif; padding: 6px; left: ${box.left}px; top: ${box.bottom + 6}px`;
      const keys = document.createElement('input');
      keys.style.cssText = 'width: 1px; height: 1px; opacity: 0; border: 0; padding: 0';
      const say = (text, answers) => {
        panel.textContent = text + ' ';
        for (const answer of answers) {
          const button = document.createElement('button');
          button.textContent = answer;
          button.dataset.qutebrowserMcpAnswer = answer;
          button.style.cssText = 'background: white; color: #800080; border: 0; font: bold 14px sans-serif; margin-left: 6px';
          panel.appendChild(button);
        }
        panel.appendChild(keys);
        keys.focus({preventScroll: true});
      };
      const path = [];
      let depth = 0, trusting = null;
      const before = document.activeElement;
      const types = ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click', 'focusin', 'focusout', 'blur',
                     'keydown', 'keypress', 'keyup', 'input'];
      const answered = answer => {
        this.state.dismiss = null;
        for (const type of types) window.removeEventListener(type, guard, true);
        panel.remove();
        if (trusting) trusting.frame.remove();
        if (before) before.focus({preventScroll: true});
        resolve(answer);
      };
      const pressed = key => {
        if (key == 's') return element.scrollIntoView({block: 'center'});
        if (key == 'n') return answered(false);
        if (!trusting) {
          if (key == 'y') return answered(true);
          if (key != 't') return;
          for (let part = element; part && part != document.documentElement; part = part.parentElement) path.push(part);
          trusting = this.frame(element, 'dashed #800080', 2147483646);
        } else {
          if (key == '+' && depth < path.length - 1) depth++;
          if (key == '-' && depth > 0) depth--;
          if (key == 'y') {
            this.keepTrusted(path, depth);
            return answered(true);
          }
          trusting.frame.remove();
          trusting = this.frame(path[depth], 'dashed #800080', 2147483646);
        }
        say('trust: + wider, - narrower, y trust, n refuse', ['+', '-', 'y', 'n']);
      };
      const guard = event => {
        if (!panel.contains(event.target) && !panel.contains(event.relatedTarget)) return;
        event.stopImmediatePropagation();
        if (!['focusin', 'focusout', 'blur'].includes(event.type)) event.preventDefault();
        if (!event.isTrusted) return;
        const button = event.target.closest && event.target.closest('[data-qutebrowser-mcp-answer]');
        if (event.type == 'click' && button) pressed(button.dataset.qutebrowserMcpAnswer);
        if (event.type == 'keydown') pressed(event.key);
      };
      if (this.state.dismiss) this.state.dismiss();
      this.state.dismiss = () => answered(null);
      for (const type of types) window.addEventListener(type, guard, true);
      document.documentElement.appendChild(panel);
      say(question, ['y', 'n', 't', 's']);
    });
  },

  async highlighted(selector, doing, value, ask) {
    this.leave();
    const element = this.element(selector);
    element.scrollIntoView({block: 'center'});
    const {frame, place} = this.frame(element, `solid ${this.colours[doing]}`, 2147483647);
    try {
      if (!ask) await this.sleep(this.highlight);
      else if (!element.closest('[data-qutebrowser-mcp-trusted]')) {
        const answer = await this.ask(element, `${value}?`);
        if (answer === null) throw new Error('the question was withdrawn, as another took its place');
        if (!answer) throw new Error(`you refused: ${doing} ${value}, at ${selector}`);
      }
    } catch (error) {
      frame.remove();
      throw error;
    }
    this.leaving(frame, place);
    return element;
  },

  leaving(frame, place) {
    const timer = setTimeout(() => this.leave(), this.left);
    window.addEventListener('scroll', place, true);
    window.addEventListener('resize', place);
    this.state.leave = () => {
      clearTimeout(timer);
      window.removeEventListener('scroll', place, true);
      window.removeEventListener('resize', place);
      frame.remove();
      delete this.state.leave;
    };
  },

  leave() {
    if (this.state.leave) this.state.leave();
  },

  titleActing(acting) {
    const mark = '🤖';
    const keep = () => {
      if (this.state.acting && !document.title.startsWith(mark))
        document.title = `${mark} ${document.title || location.host + location.pathname}`;
    };
    this.state.acting = acting;
    if (acting && !this.state.titleWatch) {
      this.state.titleWatch = new MutationObserver(keep);
      this.state.titleWatch.observe(document.head || document.documentElement,
                                    {subtree: true, childList: true, characterData: true});
    }
    if (acting) keep();
    else while (document.title.startsWith(mark)) document.title = document.title.slice(mark.length).trimStart();
  },

  async show(selector, milliseconds) {
    this.leave();
    const element = this.element(selector);
    element.scrollIntoView({block: 'center'});
    const {frame} = this.frame(element, 'solid #0000ff', 2147483647);
    await this.sleep(milliseconds);
    frame.remove();
  },

  setAsker(asker) {
    this.state.asker = asker;
  },

  dismiss(asker) {
    if (this.state.asker == asker && this.state.dismiss) this.state.dismiss();
  },

  clickable(selector) {
    const element = this.element(selector);
    if (this.shown(element)) return;
    for (let section = element; section; section = section.parentElement) {
      const opener = section.id && `[aria-controls~=${JSON.stringify(section.id)}]`;
      if (opener && document.querySelector(opener))
        throw new Error(`${selector} is not shown, being in a closed section, which ${opener} opens`);
    }
    throw new Error(`${selector} is not shown, so no mouse reaches it`);
  },

  async click(selector, ask) {
    await this.highlighted(selector, 'click', '', ask);
  },

  async fill(selector, text, ask) {
    const element = await this.highlighted(selector, 'fill', `« ${text} »`, ask);
    Object.getOwnPropertyDescriptor(Object.getPrototypeOf(element), 'value').set.call(element, text);
    element.dispatchEvent(new Event('input', {bubbles: true}));
  },

  named: (element, name) => [element, ...element.querySelectorAll('*')]
    .some(part => part.textContent.trim().toLowerCase() == name.toLowerCase()),

  async selectPlain(selector, name, ask) {
    const element = await this.highlighted(selector, 'select', `« ${name} »`, ask);
    const option = [...element.options].find(option => this.named(option, name));
    if (!option) throw new Error(`${selector} has no option ${name}`);
    element.value = option.value;
    element.dispatchEvent(new Event('input', {bubbles: true}));
    element.dispatchEvent(new Event('change', {bubbles: true}));
  },

  async openList(selector, name, ask) {
    await this.highlighted(selector, 'select', `« ${name} »`, ask);
  },

  shownOptions() {
    return [...document.querySelectorAll('[role=option], [role=menuitem], [role=menuitemcheckbox], [role=menuitemradio]')]
      .filter(element => this.shown(element));
  },

  async markOption(selector, name, milliseconds) {
    const found = () => this.shownOptions().find(option => this.named(option, name));
    if (!await this.until(found, milliseconds))
      throw new Error(`${selector} opened no shown option ${name}; it shows: `
                      + this.shownOptions().map(option => option.textContent.trim()).join(' | '));
    for (const option of document.querySelectorAll('[data-qutebrowser-mcp-option]'))
      option.removeAttribute('data-qutebrowser-mcp-option');
    found().setAttribute('data-qutebrowser-mcp-option', '');
  },

  async taken(selector, name, milliseconds) {
    const trigger = this.element(selector);
    const option = document.querySelector('[data-qutebrowser-mcp-option]');
    const took = () => option?.getAttribute('aria-selected') == 'true'
      || trigger.textContent.toLowerCase().includes(name.toLowerCase())
      || (option?.getAttribute('role') || '').startsWith('menuitem') && !this.shown(option);
    if (!await this.until(took, milliseconds))
      throw new Error(`${name} was clicked, yet ${selector} neither shows it nor marks it selected`);
  },

  async prepareUpload(selector, path, ask) {
    const element = await this.highlighted(selector, 'upload', path, ask);
    this.state.files = new Map([...document.querySelectorAll('input[type=file]')].map(input => [input, input.value]));
    element.focus();
  },

  async chosen(selector, milliseconds) {
    const chosen = () => [...document.querySelectorAll('input[type=file]')]
      .some(input => input.value && input.value != this.state.files.get(input));
    if (!await this.until(chosen, milliseconds))
      throw new Error(`qutebrowser pressed space on ${selector}, yet no file input took a file within `
                      + `${milliseconds / 1000} seconds: name the file input, or the element that opens its chooser`);
  },

  shownIn(selector) {
    if (![...document.querySelectorAll(selector)].some(element => this.shown(element)))
      throw new Error(`${location.href}, titled ${document.title}`);
  },

  markLeft(left) {
    document.documentElement.dataset.qutebrowserMcpLeft = left;
  },

  loaded(left) {
    if (document.documentElement.dataset.qutebrowserMcpLeft == left) throw new Error('the page left is still shown');
    if (document.readyState == 'loading') throw new Error('the page is still loading');
  },

  settleFrom() {
    performance.setResourceTimingBufferSize(100000);
    this.state.settleFrom = performance.now();
  },

  settled(requested, quiet) {
    if (!this.state.stirring) {
      this.state.stirred = Date.now();
      this.state.stirring = new MutationObserver(records => {
        if (records.some(record => !(record.attributeName || '').startsWith('data-qutebrowser-mcp')))
          this.state.stirred = Date.now();
      });
      this.state.stirring.observe(document.documentElement,
                                  {subtree: true, childList: true, attributes: true, characterData: true});
    }
    const ended = performance.getEntriesByType('resource')
      .filter(entry => entry.startTime >= (this.state.settleFrom || 0)).map(entry => entry.name);
    const pending = requested.filter(url => {
      const at = ended.indexOf(url);
      if (at >= 0) ended.splice(at, 1);
      return at < 0;
    });
    if (pending.length) this.state.stirred = Date.now();
    if (document.readyState == 'loading') throw new Error('it is still loading');
    if (pending.length) throw new Error(`it still waits for ${pending.join(', ')}`);
    if (Date.now() - this.state.stirred < quiet) throw new Error('it still changes');
  },
};
