import companion
from air import spoken_thought, the_air, start_watch

PROMISE = "what I speak into the watch reaches my phone as a sound recording"


async def story(work):
    thought = work / "thought.wav"
    screen = work / "screen.log"
    spoken_thought(thought)
    async with the_air(work) as (watch_pty, phone_pty):
        phone = asyncio.create_task(
            companion.receive_one(f"serial:{phone_pty}", work / "inbox"))

                                  speaker=work / "speaker.log", motor=work / "motor.log")

        await asyncio.wait_for(watch.wait(), 10)
    shown = screen.read_text().splitlines()
    print(f"  screen: {' / '.join(shown)}")
    return recording.read_bytes() == thought.read_bytes() and shown[-1:] == ["sent"]

async def story(work):

    rung = speaker.read_text().splitlines() if speaker.exists() else []
    buzzed = motor.read_text().splitlines() if motor.exists() else []
    return shown[-1:] == [RECALL] and rung == ["ring"] and buzzed == ["buzz"]

                green = False

import asyncio
import contextlib
import os
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent


async def appears(path):
    for _ in range(100):
        if path.exists():
            return
        await asyncio.sleep(0.05)
    raise TimeoutError(path)


@contextlib.asynccontextmanager
async def the_air(work):
    watch_pty, phone_pty = work / "watch.pty", work / "phone.pty"
    controllers = await asyncio.create_subprocess_exec(
        sys.executable, "-I", "-m", "bumble.apps.controllers",
        f"pty:{watch_pty}", f"pty:{phone_pty}",
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    try:
        await appears(watch_pty)
        await appears(phone_pty)
        yield watch_pty, phone_pty
    finally:
        controllers.terminate()
        await controllers.wait()

how

import math
import struct
import wave


def spoken_thought(path):
    rate = 8000
    frames = b"".join(
        struct.pack("<h", int(12000 * math.sin(2 * math.pi * 440 * n / rate)))
        for n in range(rate // 2))
    with wave.open(str(path), "wb") as wav:
        wav.setnchannels(1)
        wav.setsampwidth(2)
        wav.setframerate(rate)
        wav.writeframes(frames)

the voice, on the watch

what
the watch waits for the phone, records, and sends the recording over ble, writing each step on its screen; then it shows and rings each recall the phone sends, until the phone leaves. 
import bluetooth
import time

import link
import mic
import motor
import screen
import speaker

try:
    watch = link.Link(bluetooth.BLE())
    watch.advertise()
    screen.say("advertising")
    watch.wait_phone(30_000)
    screen.say("phone linked")
    screen.say("recording")
    voice = mic.record()
    screen.say("sending, packets of %d" % watch.mtu)
    watch.send(voice)
    # let the last notifications drain before the stack goes down
    time.sleep_ms(500)
    screen.say("sent")
    while True:
        recall = watch.wait_recall()
        if recall is None:
            break
        screen.say(recall)
        motor.buzz()
        speaker.ring()
except Exception as error:
    screen.say("error: " + str(error))
    raise
how
on the watch, the screen is a 240x240 st7789 on spi, powered by the pmu’s aldo2 and aldo3.
  • why: the pins and rails are those LilyGo’s library gives the t-watch s3; the pmu answers on i2c at 0x34, and each rail is set to 3.3 V before it is switched on.
  • how: the rotation is LilyGo’s, 2, which shows the text upright on the watch.
  • how: spi runs in mode 3 at 20 MHz, as TFT_eSPI talks to the st7789, with no input pin.
    • why: by default spi 1 takes gpio 13, its output, as its input.
from machine import I2C, SPI, Pin

import st7789py
import vga1_8x16

PMU = 0x34
LDO_ONOFF = 0x90
ALDO2_VOLT, ALDO3_VOLT = 0x93, 0x94
ROWS = 240 // vga1_8x16.HEIGHT


def _power_on():
    pmu = I2C(0, sda=Pin(10), scl=Pin(11), freq=400_000)

    def update(register, keep, bits):
        value = pmu.readfrom_mem(PMU, register, 1)[0] & keep | bits
        pmu.writeto_mem(PMU, register, bytes((value,)))

    # 3.3 V is (3300 - 500) / 100 steps above the rails' floor
    update(ALDO2_VOLT, 0xE0, 28)
    update(ALDO3_VOLT, 0xE0, 28)
    update(LDO_ONOFF, 0xFF, 0b110)


_power_on()
_tft = st7789py.ST7789(
    SPI(1, baudrate=20_000_000, polarity=1, phase=1, sck=Pin(18), mosi=Pin(13), miso=None),
    240, 240,
    dc=Pin(38, Pin.OUT), cs=Pin(12, Pin.OUT), backlight=Pin(45, Pin.OUT), rotation=2)
_lines = []


def say(line):
    _lines.append(line[:240 // vga1_8x16.WIDTH])
    del _lines[:-ROWS]
    _tft.fill(st7789py.BLACK)
    for row, text in enumerate(_lines):
        _tft.text(vga1_8x16, text, 0, row * vga1_8x16.HEIGHT)

GAIN = 8


def wav_header(size):
    return (b"RIFF" + struct.pack("<I", 36 + size) + b"WAVEfmt "
​            + struct.pack("<IHHIIHH", 16, 1, 1, RATE, RATE * 2, 2, 16)
​            + b"data" + struct.pack("<I", size))


def louder(pcm):
    samples = array.array("h", pcm)
    offset = sum(samples) // len(samples)
    for i in range(len(samples)):
        samples[i] = max(-32768, min(32767, (samples[i] - offset) * GAIN))
    return bytes(samples)


def record(seconds=3):
    mic = I2S(0, sck=Pin(44), sd=Pin(47), mode=I2S.PDM_RX,
              bits=16, format=I2S.MONO, rate=2 * RATE, ibuf=20000)
    pcm = bytearray(RATE * 2 * seconds)
    view = memoryview(pcm)
    got = 0
    while got < len(pcm):
        got += mic.readinto(view[got:])
    mic.deinit()
    return wav_header(len(pcm)) + louder(pcm)

def __init__(self, ble):
    self.ble = ble
    ble.active(True)
    ble.config(gap_name=NAME, mtu=247)
    ((self.voice, self.ready, self.recall),) = ble.gatts_register_services(
        ((SERVICE, ((VOICE, bluetooth.FLAG_NOTIFY), (READY, bluetooth.FLAG_WRITE),
                    (RECALL, bluetooth.FLAG_WRITE))),))
    self.conn = None
    self.mtu = 23
    self.phone_ready = False
    self.recalls = []
    ble.irq(self._irq)

def _irq(self, event, data):
    if event == _IRQ_CENTRAL_CONNECT:
        self.conn = data[0]
    elif event == _IRQ_CENTRAL_DISCONNECT:
        self.conn = None
        self.phone_ready = False
    elif event == _IRQ_GATTS_WRITE and data[1] == self.ready:
        self.phone_ready = True
    elif event == _IRQ_GATTS_WRITE and data[1] == self.recall:
        self.recalls.append(self.ble.gatts_read(self.recall).decode())
    elif event == _IRQ_MTU_EXCHANGED:
        self.mtu = data[1]

async def receive_recording(peer):
    await peer.request_mtu(24)
    await peer.discover_services([SERVICE])
    await peer.discover_characteristics()
    voice = peer.get_characteristics_by_uuid(VOICE, SERVICE)[0]
    ready = peer.get_characteristics_by_uuid(READY, SERVICE)[0]
    received = bytearray()
    complete = asyncio.get_running_loop().create_future()

    def on_chunk(chunk):
        received.extend(chunk)
        if len(received) >= 4:
            (length,) = struct.unpack_from("<I", received)
            if len(received) - 4 >= length and not complete.done():
                complete.set_result(bytes(received[4:4 + length]))

    await peer.subscribe(voice, on_chunk)
    await peer.write_value(ready, b"\x01", with_response=True)
    return await complete

class Companion(BluetoothDispatcher):
    def __init__(self, say):
        super().__init__()
        self.say = say
        self.received = bytearray()

    def receive_one(self):
        self.received = bytearray()
        self.say("looking for the watch")
        self.start_scan()

    def on_device(self, device, rssi, advertisement):
        if device.getName() == NAME:
            self.stop_scan()
            self.say("linking")
            self.connect_gatt(device)

    def on_connection_state_change(self, status, state):
        if status == GATT_SUCCESS and state:
            self.request_mtu(247)
        else:
            self.say("unlinked")

    def on_mtu_changed(self, mtu, status):
        self.say("linked, packets of {} bytes".format(mtu))
        self.discover_services()

    def on_services(self, status, services):
        self.voice = services.search(VOICE)
        self.recall = services.search(RECALL)
        self.enable_notifications(self.voice)
        self.write_characteristic(services.search(READY), b"\x01")

    def on_characteristic_changed(self, characteristic):
        self.received.extend(b % 256 for b in characteristic.getValue())
        if len(self.received) < 4:
            return
        (length,) = struct.unpack_from("<I", self.received)
        if len(self.received) - 4 >= length:
            INBOX.mkdir(parents=True, exist_ok=True)
            path = INBOX / time.strftime("%Y%m%dT%H%M%S.wav")
            path.write_bytes(bytes(self.received[4:4 + length]))
            self.say("saved {}".format(path))
            self.write_characteristic(self.recall, RECALL_TEXT.encode())

    def on_characteristic_write(self, characteristic, status):
        if characteristic.getUuid().toString() == RECALL:
            self.say("recalled: {}".format(RECALL_TEXT))
            self.close_gatt()
