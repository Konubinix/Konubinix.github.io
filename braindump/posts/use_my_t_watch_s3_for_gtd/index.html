import asyncio
import contextlib
import os
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent


async def appears(path):
    for _ in range(100):
        if path.exists():
            return
        await asyncio.sleep(0.05)
    raise TimeoutError(path)


@contextlib.asynccontextmanager
async def the_air(work):
    watch_pty, phone_pty = work / "watch.pty", work / "phone.pty"
    controllers = await asyncio.create_subprocess_exec(
        sys.executable, "-I", "-m", "bumble.apps.controllers",
        f"pty:{watch_pty}", f"pty:{phone_pty}",
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    try:
        await appears(watch_pty)
        await appears(phone_pty)
        yield watch_pty, phone_pty
    finally:
        controllers.terminate()
        await controllers.wait()

import math

import wave


def spoken_thought(path):
    rate = 8000
    frames = b"".join(

    client, _ = server.accept()
    client.setblocking(False)
    client.setsockopt(socket.SOL_SOCKET, _CALLBACK, _typed)

    return time.ticks_diff(time.ticks_ms(), _used)

what
the watch waits for the phone, then, while it is linked, shows each recall the phone sends, and records from a tap on the screen to the next, 2 min at most, then sends over ble, writing each step on it; when the phone leaves, it waits for it again.
how
a recall’s priority is ntfy’s, 1 to 5: 1 is written on a dark screen without lighting it, 2 lights it, 3 also buzzes once, 4 buzzes and rings, 5 buzzes and rings three times.
  • how: 1 and 2 are for senders too noisy to be heard each time.
how
while a recording goes out, its line counts « sending 45 % », redone in place each 5 %, then becomes « sent in 12 s »; the garbage it left is then collected.
  • how: as a send starts, the screen turns read-only, and stays so; the wheel makes it touchable again.
    • why: a tap, or a stray pulse of the panel, during the send would start another recording.
how
a send that fails says « send stalled, kept », and the recording waits for the next link.
how
a recall’s line starts with when the phone took it, « 08:42 », or « 22/10 08:42 » when not today.
how
unlinked, a recording is kept and sent at the next link; while one is kept, a tap records nothing more.
  • why: more would fill the ram.
how
the screen’s bottom row is a button that switches the wifi on and off; it also shows the battery’s percent, blue while charging, its voltage and the ram free, « 95% 4.06V wifi on 3.2M », read again each minute the screen is lit, and at each wake.
how
once joined, the wifi’s address is said as a line.
how
a tap on a dark screen lights it read-only, where taps do nothing; the wheel lights it touch-sensitive, dark or read-only, and darkens it when touch-sensitive.
how
between looks, the watch naps 50 ms while lit and 250 ms while dark.
how
touch-sensitive, a finger dragged down shows older lines, a row per 16 px, and dragged up newer ones; the wheel’s turn sends nothing, so it can’t.
how
each hour, the watch adds its seconds since boot, percent and millivolts to battery.log.
how

how

def _sum(samples: ptr16, count: int) -> int:
    total = 0

        value = int(samples[i])
        if value > 32767:
            value -= 65536

@micropython.viper
def _louder(samples: ptr16, count: int, offset: int, gain: int):
    for i in range(count):
        value = int(samples[i])
        if value > 32767:
            value -= 65536
        value = (value - offset) * gain
        if value > 32767:
            value = 32767
        if value < -32768:
            value = -32768
        samples[i] = value & 0xFFFF


def louder(pcm):
    count = len(pcm) // 2
    total = 0
    for start in range(0, count, SUM_CHUNK):
        total += _sum(pcm[2 * start:], min(SUM_CHUNK, count - start))
    _louder(pcm, count, total // count, GAIN)

    got = 0
    second = 0
    while HEADER + got < end:

        if stop is not None and stop():
            break
        if tick is not None and got // (RATE * 2) != second:
            second = got // (RATE * 2)
            tick(second)

    got -= got % 2
    louder(view[HEADER:HEADER + got])
    view[:HEADER] = wav_header(got)
    return view[:HEADER + got]

def send(self, payload, progress=None):
    step = self.mtu - 3
    view = memoryview(payload)
    told = 0
    self._notify(struct.pack("<I", len(payload)) + bytes(view[:step - 4]))
    for offset in range(step - 4, len(payload), step):
        self._notify(view[offset:offset + step])
        percent = offset * 100 // len(payload) // PROGRESS_STEP * PROGRESS_STEP
        if progress is not None and percent != told:
            told = percent
            progress(percent)

import micropython
from micropython import const

_MP_STREAM_POLL = const(3)
_MP_STREAM_POLL_RD = const(1)


class Stream(io.IOBase):
    def __init__(self, link):
        self.link = link
        self.inbox = bytearray()
        self.outbox = bytearray()

    def received(self, data):
        self.inbox += data
        os.dupterm(self)
        os.dupterm_notify(None)

    def readinto(self, buf):
        if not self.inbox:
            return None
        count = min(len(buf), len(self.inbox))
        buf[:count] = self.inbox[:count]
        self.inbox = self.inbox[count:]
        return count

    def ioctl(self, op, arg):
        if op == _MP_STREAM_POLL and self.inbox:
            return _MP_STREAM_POLL_RD
        return 0

    def write(self, buf):
        idle = not self.outbox
        self.outbox += buf
        if idle:
            micropython.schedule(self._flush, None)
        return len(buf)

    def _flush(self, _):
        step = self.link.mtu - 3
        chunk, self.outbox = self.outbox[:step], self.outbox[step:]
        try:
            self.link._notify(chunk, self.link.repl_out)
        except Exception:
            self.outbox = bytearray()
        if self.outbox:
            micropython.schedule(self._flush, None)

class Companion(BluetoothDispatcher):
    def __init__(self, say):
        super().__init__()
        self.say = say
        self.received = bytearray()
        self.mtu = 23
        self.recall = None
        self.discovering = False
        self.recalls = deque()
        self.recalling = None
        self.repl_in = None
        self.steps = deque()
        self.to_watch = deque()
        self.writing = False
        self.lock = threading.Lock()
        self.desktop = None
        self.draining = False

    def receive_one(self):
        self.received = bytearray()
        self.linking = False
        watch = connected_watch()
        if watch is not None:
            self.on_device(watch, None, None)
            return
        self.say("looking for the watch")
        self.start_scan(filters=[DeviceNameFilter(NAME)])

    def on_bluetooth_adapter_state_change(self, state):
        if state == ADAPTER_ON:
            self.say("bluetooth back")
            threading.Timer(ADAPTER_WAIT_S, self.receive_one).start()

    def on_device(self, device, rssi, advertisement):
        if device.getName() == NAME and not self.linking:
            self.linking = True
            self.stop_scan()
            self.say("linking")
            self.close_gatt()
            self.connect_gatt(device)

    def on_connection_state_change(self, status, state):
        if status == GATT_SUCCESS and state:
            self.discovering = False
            self.request_mtu(247)
            threading.Timer(MTU_WAIT_S, self.discover, ("no packet size answer",)).start()
        else:
            self.say("unlinked")
            with self.lock:
                self.recall = self.recalling = self.repl_in = None
            self.steps.clear()
            self.close_gatt()
            self.receive_one()

    def on_mtu_changed(self, mtu, status):
        self.mtu = mtu
        self.discover("linked, packets of {} bytes".format(mtu))

    def discover(self, why):
        with self.lock:
            if self.discovering:
                return
            self.discovering = True
        self.say(why)
        self.discover_services()

    def on_services(self, status, services):
        self.voice = services.search(VOICE)
        self.saved = services.search(SAVED)
        self.services = services
        self.gatt.requestConnectionPriority(LOW_POWER)
        self.steps = deque([
            lambda: self.enable_notifications(self.voice),
            lambda: self.enable_notifications(services.search(REPL_OUT)),
            lambda: self.write_characteristic(services.search(TIME), local_time()),
            lambda: self.write_characteristic(services.search(READY), phone_name()),
        ])
        self.next_step()

    def next_step(self):
        if self.steps:
            self.steps.popleft()()
            return
        self.say("ready")
        with self.lock:
            self.recall = self.services.search(RECALL)
            self.repl_in = self.services.search(REPL_IN)
        self.send_recalls()
        self.pump()

    def on_descriptor_write(self, descriptor, status):
        self.next_step()

    def on_characteristic_changed(self, characteristic):
        if characteristic.getUuid().toString() == REPL_OUT:
            if self.desktop is not None and not self.draining:
                try:
                    self.desktop.sendall(bytes(b % 256 for b in characteristic.getValue()))
                except OSError as error:
                    self.say("relay cut: {}".format(error))
            return
        if not self.received:
            self.gatt.requestConnectionPriority(HIGH)
        self.received.extend(b % 256 for b in characteristic.getValue())
        if len(self.received) < 4:
            return
        (length,) = struct.unpack_from("<I", self.received)
        if len(self.received) - 4 >= length:
            INBOX.mkdir(parents=True, exist_ok=True)
            path = INBOX / time.strftime("%Y%m%dT%H%M%S.wav")
            path.write_bytes(bytes(self.received[4:4 + length]))
            self.say("saved {}".format(path))
            self.received = bytearray()
            self.gatt.requestConnectionPriority(LOW_POWER)
            self.write_characteristic(self.saved, b"\x01")

    def on_characteristic_write(self, characteristic, status):
        if characteristic.getUuid().toString() == SAVED:
            self.say("told the watch")
        elif characteristic.getUuid().toString() in (TIME, READY):
            self.next_step()
        elif characteristic.getUuid().toString() == REPL_IN:
            with self.lock:
                self.writing = False
            self.pump()
        elif characteristic.getUuid().toString() == RECALL:
            with self.lock:
                sent, self.recalling = self.recalling, None
                if status == GATT_SUCCESS and self.recalls and self.recalls[0] is sent:
                    self.recalls.popleft()
            if status == GATT_SUCCESS and sent is not None:
                self.say("recall given: {}".format(sent["text"]))
            self.send_recalls()

    def queue_recall(self, text, priority, expires_h):
        with self.lock:
            self.recalls.append({"text": text, "priority": priority, "sent": local_minute(),
                                 "expires": time.time() + expires_h * 3600})
        self.say("recall queued, priority {}: {}".format(priority, text))
        self.send_recalls()

    def send_recalls(self):
        with self.lock:
            while self.recalls and self.recalls[0]["expires"] < time.time():
                self.say("recall expired: {}".format(self.recalls.popleft()["text"]))
            if self.recall is None or self.recalling is not None or not self.recalls:
                return
            self.recalling = self.recalls[0]
            payload = struct.pack("<BHBBBB", self.recalling["priority"], *self.recalling["sent"])
            payload += self.recalling["text"].encode()
            recall = self.recall
        try:
            self.write_characteristic(recall, payload)
        except Exception as error:
            self.say("recall kept for the next link: {}".format(error))
            with self.lock:
                self.recalling = None

    def relay(self, data):
        step = self.mtu - 3
        with self.lock:
            self.to_watch.extend(data[i:i + step] for i in range(0, len(data), step))
        self.pump()

    def pump(self):
        with self.lock:
            if self.writing or not self.to_watch or self.repl_in is None:
                return
            self.writing = True
            chunk = self.to_watch.popleft()
        self.write_characteristic(self.repl_in, chunk)

    def serve_relay(self):
        server = socket.socket()
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server.bind(("0.0.0.0", RELAY_PORT))
        server.listen(1)
        while True:
            self.desktop, address = server.accept()
            self.say("relaying the REPL for {}".format(address[0]))
            self.draining = True
            self.relay(b"\r\x03\x03")
            time.sleep(DRAIN_S)
            self.draining = False
            try:
                while True:
                    data = self.desktop.recv(1024)
                    if not data:
                        break
                    self.relay(data)
            except OSError as error:
                self.say("relay cut: {}".format(error))
            self.desktop.close()
            self.desktop = None
            self.say("relay closed")
