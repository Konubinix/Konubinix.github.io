Or, using using a python2 stack

import os
from rpyc.core import SlaveService
from rpyc.utils.server import ForkingServer

def run(*args, **kwargs):
    print("Starting rpyc")
    os.a = (locals(), globals(), args, kwargs)
    os.environ["NO_COLOR"] = "1"
    ForkingServer(
        SlaveService,
        hostname="0.0.0.0",
        port=9999,
        reuse_addr=True,
    ).start()
