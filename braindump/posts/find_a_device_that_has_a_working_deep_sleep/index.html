find a device that has a working deep sleep

import machine
import time

def deep_sleep(msecs):
    rtc = machine.RTC()
    rtc.irq(trigger=rtc.ALARM0, wake=machine.DEEPSLEEP)
    rtc.alarm(rtc.ALARM0, msecs)
    machine.deepsleep()

print("hello")
time.sleep(2)
print("to sleep")
deep_sleep(2000)
