Application script

ES module containing all application logic: service worker registration, utilities, Petite-Vue app state, SyncedStore sync, and media handling.

All custom events (pleaseplay, gotonext, etc.) are dispatched on window and caught via window.addEventListener (Petite-Vue has no .window modifier).

The medium x-bind pattern from Alpine is replaced by setupMedium(el) which attaches all event listeners to the media element in the @vue:mounted hook.

Dir/podcast extended state is hoisted to expandedDirs / expandedPodcasts maps in the App (Petite-Vue has no Alpine.$data(el), so the locate button needs top-level access).

Persistence uses v-effect calling persistState() / pushToYjs() on hidden <span> elements (replaces Alpine $persist and $watch).

SyncedStore replaces raw Yjs for settings, current state, and bookmark maps. Episode times keep raw Yjs via getYjsDoc(store).getMap("episode-times") for the Math.max merge strategy (SyncedStore only supports LWW).

// --- Helpers ---

function formatTime(seconds) {
    seconds = Math.floor(seconds)
    let hours = Math.floor(seconds / 3600);
    let minutes = Math.floor((seconds % 3600) / 60);
    let remainingSeconds = seconds % 60;
    let parts = [];
    if (hours > 0) {
        parts.push(hours.toString().padStart(2, '0'));
    }
    parts.push(minutes.toString().padStart(2, '0'));
    parts.push(remainingSeconds.toString().padStart(2, '0'));
    return parts.join(':');
}

function loadFromLS(key, fallback) {
    try {
        const v = localStorage.getItem(key)
        return v !== null ? JSON.parse(v) : fallback
    } catch { return fallback }
}

function saveToLS(key, value) {
    localStorage.setItem(key, JSON.stringify(value))
}

function emit(name, detail) {
    window.dispatchEvent(new CustomEvent(name, { detail }))
}

function App() {
    // Non-reactive closure state for Yjs (avoids Vue wrapping SyncedStore proxies)
    let _store = null
    let _timesMap = null
    let _yjsTimeSyncTimeout = null
    let _localChangeTs = {}
    let _mediaEl = null

    return {
        src: "podcasts.json",
        dirs: undefined,

        dark_mode: loadFromLS("dark_mode", false),
        show_old: loadFromLS("show_old", false),
        auto_next: loadFromLS("auto_next", true),

        podcast_last_episode: loadFromLS("podcast_last_episode3", {}),
        dir_last_episode: loadFromLS("dir_last_episode", {}),

        current_speed: null,
        current_progress: {cur: 0, duration: 0},
        current: {
            playing: false,
            episode: loadFromLS("episode", null),
            podcast: loadFromLS("podcast", null),
            dir: loadFromLS("dir", null),
        },

        expandedDirs: {},
        expandedPodcasts: {},

        _deferredInstallPrompt: null,
        _yjsSyncing: false,
        _yjsConnected: false,
        _yjsInitialSyncDone: false,

        // --- Init ---

        async init_app() {
            try {
                const res = await fetch(`./${this.src}`, {cache: 'no-store'})
                this.dirs = await res.json()
                var lastpodcast
                var lastdir
                var keepcurrent = false
                var keeplastdir
                var keeplastpodcast
                for (const dir in this.dirs) {
                    lastdir = this.dir_last_episode[dir]
                    keeplastdir = false
                    for (const podcast of this.dirs[dir]) {
                        lastpodcast = this.podcast_last_episode[podcast.name]
                        keeplastpodcast = false
                        for (const episode of podcast.episodes) {
                            if (lastdir && episode.url === lastdir.episode.url && podcast.name === lastdir.podcast.name) {
                                keeplastdir = true
                            }
                            if (lastpodcast && episode.url === lastpodcast.episode.url) {
                                keeplastpodcast = true
                            }
                            if (this.current.episode && this.current.episode.url === episode.url) {
                                keepcurrent = true
                            }
                        }
                        if (lastpodcast && !keeplastpodcast) {
                            delete this.podcast_last_episode[podcast.name]
                        }
                    }
                    if (lastdir && !keeplastdir) {
                        delete this.dir_last_episode[dir]
                    }
                }
                // Clean stale entries for podcasts/dirs no longer in the feed
                let allPodcastNames = new Set()
                for (const dir in this.dirs) {
                    for (const podcast of this.dirs[dir]) {
                        allPodcastNames.add(podcast.name)
                    }
                }
                for (const name in this.podcast_last_episode) {
                    if (!allPodcastNames.has(name)) {
                        delete this.podcast_last_episode[name]
                    }
                }
                for (const dir in this.dir_last_episode) {
                    if (!(dir in this.dirs)) {
                        delete this.dir_last_episode[dir]
                    }
                }
                if (!keepcurrent) {
                    this.current.episode = null
                    if (this.current.podcast && !allPodcastNames.has(this.current.podcast.name)) {
                        this.current.podcast = null
                        this.current.dir = null
                    }
                }
                if (!this.current.episode && this.current.podcast) {
                    let last = this.podcast_last_episode[this.current.podcast.name]
                    if (last) {
                        this.current.episode = last.episode
                    }
                }

                // Set up window-level event handlers for gotonext/gotoprev
                const self = this
                window.addEventListener('gotonext', (e) => {
                    if (e.detail?.playing) self.current.playing = true
                    self.goto_next()
                })
                window.addEventListener('gotoprev', (e) => {
                    if (e.detail?.playing) self.current.playing = true
                    self.goto_prev()
                })

                this._setupSync()
                window.addEventListener('beforeinstallprompt', (e) => {
                    e.preventDefault()
                    this._deferredInstallPrompt = e
                })
                window.addEventListener('appinstalled', () => {
                    this._deferredInstallPrompt = null
                })
            }
            catch(err) {
                error(err)
            }
        },

        // --- Navigation ---

        key(stuff) {
            return `${stuff}_${this.current.episode.url}`
        },

        goto_next() {
            localStorage.setItem(this.key('time'), 0)
            this._syncEpisodeTime(this.current.episode.url, 0)
            let pos = this.current.podcast.episodes.findIndex(
                (episode) => episode.url === this.current.episode.url
            )
            if (pos + 1 === this.current.podcast.episodes.length) {
                localStorage.setItem(this.current.podcast.name + "_looped_first_episode", this.current.podcast.episodes[0].url)
            }
            this.current.episode = this.current.podcast.episodes[(pos + 1) % this.current.podcast.episodes.length]
        },

        goto_prev() {
            localStorage.setItem(this.key('time'), 0)
            this._syncEpisodeTime(this.current.episode.url, 0)
            let pos = this.current.podcast.episodes.findIndex(
                (episode) => episode.url === this.current.episode.url
            )
            this.current.episode = this.current.podcast.episodes[(pos - 1 + this.current.podcast.episodes.length) % this.current.podcast.episodes.length]
        },

        async shareEpisode() {
            if (navigator.share) {
                try {
                    await navigator.share({
                        title: this.current.episode.name,
                        text: this.current.episode.name,
                        url: this.current.episode.origin_url,
                    });
                    info('Episode shared successfully');
                } catch (err) {
                    info('Error sharing episode: ' + err);
                    console.log(`Error sharing episode: ${err}, ${this.current.episode.origin_url}`);
                }
            } else {
                try {
                    await navigator.clipboard.writeText(this.current.episode.origin_url);
                    info('Episode URL copied to clipboard');
                } catch (err) {
                    info('Error copying to clipboard: ' + err);
                }
            }
        },

        onended() {
            localStorage.setItem(this.key('time'), 0)
            this._syncEpisodeTime(this.current.episode.url, 0)
            if (this.auto_next) {
                emit('gotonext', {playing: true})
            }
        },

        // --- Template helpers (replace inline async handlers) ---

        async playDir(dir, podcasts) {
            this.current.podcast = podcasts[0]
            this.current.dir = dir
            this.current.episode = null
            await nextTick()
            this.current.episode = podcasts[0].episodes[0]
            localStorage.setItem(`time_${this.current.episode.url}`, 0)
            await nextTick()
            emit('pleaseplay')
            emit('pleaserewind')
        },

        async resumeDir(dir) {
            let podcast = this.dir_last_episode[dir].podcast
            let episode = this.dir_last_episode[dir].episode
            this.current.podcast = podcast
            this.current.dir = dir
            this.current.episode = null
            await nextTick()
            this.current.episode = episode
            await nextTick()
            emit('pleaseplay')
            emit('pleaserewind')
        },

        async playPodcast(podcast, dir) {
            this.current.podcast = podcast
            this.current.dir = dir
            this.current.episode = null
            await nextTick()
            this.current.episode = podcast.episodes[0]
            localStorage.setItem(`time_${this.current.episode.url}`, 0)
            await nextTick()
            emit('pleaseplay')
            emit('pleaserewind')
        },

        async resumePodcast(podcast) {
            let dir = this.podcast_last_episode[podcast.name].dir
            let episode = this.podcast_last_episode[podcast.name].episode
            this.current.dir = dir
            this.current.podcast = podcast
            this.current.episode = null
            await nextTick()
            this.current.episode = episode
            await nextTick()
            emit('pleaseplay')
            emit('pleaserewind')
        },

        async selectEpisode(episode, podcast, dir) {
            this.current.episode = null
            await nextTick()
            this.current.episode = episode
            this.current.podcast = podcast
            this.current.dir = dir
            this.current.playing = true
        },

        async locateCurrent() {
            if (!this.current.dir) return
            this.expandedDirs[this.current.dir] = true
            if (this.current.podcast) {
                this.expandedPodcasts[this.current.podcast.name] = true
            }
            await nextTick()
            document.querySelector('#episode.bg-green-100')?.scrollIntoView({behavior: 'smooth', block: 'center'})
        },

        // --- Persistence (called from v-effect) ---

        saveBookmarks() {
            if (!this.current.podcast || !this.current.episode || !this.current.dir) return
            this.podcast_last_episode[this.current.podcast.name] = {episode: this.current.episode, dir: this.current.dir}
            this.dir_last_episode[this.current.dir] = {episode: this.current.episode, podcast: this.current.podcast}
        },

        persistState() {
            saveToLS('dark_mode', this.dark_mode)
            saveToLS('show_old', this.show_old)
            saveToLS('auto_next', this.auto_next)
            saveToLS('episode', this.current.episode)
            saveToLS('podcast', this.current.podcast)
            saveToLS('dir', this.current.dir)
            saveToLS('podcast_last_episode3', this.podcast_last_episode)
            saveToLS('dir_last_episode', this.dir_last_episode)
        },

        // --- Yjs sync (called from v-effect) ---

        pushToYjs() {
            if (!this._yjsInitialSyncDone || this._yjsSyncing || !_store) return
            const settings = _store["podcast-player"]
            for (const key of ['show_old', 'auto_next', 'dark_mode']) {
                const val = this[key]
                const remote = settings[key]
                if (!remote || remote.value !== val) {
                    settings[key] = {value: val, ts: Date.now()}
                }
            }
            const snap = JSON.parse(JSON.stringify({
                episode: this.current.episode,
                podcast: this.current.podcast,
                dir: this.current.dir,
            }))
            const remoteCurrent = settings.current
            if (!remoteCurrent || JSON.stringify(remoteCurrent.value) !== JSON.stringify(snap)) {
                settings.current = {value: snap, ts: Date.now()}
            }
            const podcastLastMap = _store["podcast-last-episode"]
            for (const [name, entry] of Object.entries(this.podcast_last_episode)) {
                const serialized = JSON.parse(JSON.stringify(entry))
                if (JSON.stringify(podcastLastMap[name]) !== JSON.stringify(serialized)) {
                    podcastLastMap[name] = serialized
                }
            }
            const dirLastMap = _store["dir-last-episode"]
            for (const [name, entry] of Object.entries(this.dir_last_episode)) {
                const serialized = JSON.parse(JSON.stringify(entry))
                if (JSON.stringify(dirLastMap[name]) !== JSON.stringify(serialized)) {
                    dirLastMap[name] = serialized
                }
            }
        },

        _syncEpisodeTime(url, time) {
            if (!_timesMap) return
            clearTimeout(_yjsTimeSyncTimeout)
            _yjsTimeSyncTimeout = setTimeout(() => {
                _timesMap.set(url, time)
            }, 5000)
        },

        _setupSync() {
            _store = syncedStore({
                "podcast-player": {},
                "podcast-last-episode": {},
                "dir-last-episode": {},
            })
            const doc = getYjsDoc(_store)
            _timesMap = doc.getMap("episode-times")
            const self = this

            // Track local change timestamps (for initial merge)
            // These run before sync is established, recording when local changes happen
            // We don't have @vue/reactivity watch, so we track in the push method
            // and in the setter methods. For initial merge, we use _localChangeTs
            // which is set by the persistState v-effect side.

            // --- SyncedStore -> local state observers ---

            observeDeep(_store["podcast-player"], () => {
                self._yjsSyncing = true
                const settings = _store["podcast-player"]
                for (const key of ['show_old', 'auto_next', 'dark_mode']) {
                    const wrapper = settings[key]
                    if (wrapper?.value !== undefined) self[key] = wrapper.value
                }
                const wrapper = settings.current
                if (wrapper?.value) {
                    self.current.episode = wrapper.value.episode
                    self.current.podcast = wrapper.value.podcast
                    self.current.dir = wrapper.value.dir
                }
                if (!self.current.episode && self.current.podcast) {
                    let last = self.podcast_last_episode[self.current.podcast.name]
                    if (last) self.current.episode = last.episode
                }
                setTimeout(() => { self._yjsSyncing = false }, 100)
            })

            observeDeep(_store["podcast-last-episode"], () => {
                self._yjsSyncing = true
                const map = _store["podcast-last-episode"]
                const obj = {}
                for (const key of Object.keys(map)) { obj[key] = map[key] }
                if (JSON.stringify(obj) !== JSON.stringify(self.podcast_last_episode)) {
                    self.podcast_last_episode = obj
                }
                setTimeout(() => { self._yjsSyncing = false }, 100)
            })

            observeDeep(_store["dir-last-episode"], () => {
                self._yjsSyncing = true
                const map = _store["dir-last-episode"]
                const obj = {}
                for (const key of Object.keys(map)) { obj[key] = map[key] }
                if (JSON.stringify(obj) !== JSON.stringify(self.dir_last_episode)) {
                    self.dir_last_episode = obj
                }
                setTimeout(() => { self._yjsSyncing = false }, 100)
            })

            // Episode times (raw Yjs for Math.max merge)
            _timesMap.observe((event) => {
                event.changes.keys.forEach((change, url) => {
                    const time = _timesMap.get(url)
                    if (time !== undefined) {
                        localStorage.setItem('time_' + url, time)
                    }
                })
            })

            // --- WebSocket provider ---

            const wsUrl = (location.protocol === 'https:' ? 'wss://' : 'ws://') + location.host + '/ywebsocket'
            const provider = new WebsocketProvider(
                wsUrl,
                'podcast-player',
                doc
            )
            provider.on('status', (event) => {
                self._yjsConnected = event.status === 'connected'
            })

            // --- Initial sync merge ---

            provider.on('sync', (isSynced) => {
                if (!isSynced) return
                self._yjsSyncing = true
                const settings = _store["podcast-player"]

                // Simple keys: timestamp-based merge
                for (const key of ['show_old', 'auto_next', 'dark_mode']) {
                    const remote = settings[key]
                    const localTs = _localChangeTs[key] || 0
                    const remoteTs = remote?.ts || 0
                    if (remote !== undefined && remoteTs >= localTs) {
                        self[key] = remote.value !== undefined ? remote.value : remote
                    } else {
                        settings[key] = {value: JSON.parse(JSON.stringify(self[key])), ts: localTs || Date.now()}
                    }
                }

                // Current: timestamp-based merge
                const remoteCurrent = settings.current
                const localCurrentTs = _localChangeTs['current'] || 0
                const remoteCurrentTs = remoteCurrent?.ts || 0
                if (remoteCurrent && remoteCurrentTs >= localCurrentTs && remoteCurrent.value) {
                    self.current.episode = remoteCurrent.value.episode
                    self.current.podcast = remoteCurrent.value.podcast
                    self.current.dir = remoteCurrent.value.dir
                } else {
                    settings.current = {
                        value: JSON.parse(JSON.stringify({
                            episode: self.current.episode,
                            podcast: self.current.podcast,
                            dir: self.current.dir,
                        })),
                        ts: localCurrentTs || Date.now()
                    }
                }
                if (!self.current.episode && self.current.podcast) {
                    let last = self.podcast_last_episode[self.current.podcast.name]
                    if (last) self.current.episode = last.episode
                }

                // podcast_last_episode: per-key merge via dedicated map
                const podcastLastMap = _store["podcast-last-episode"]
                for (const name of Object.keys(podcastLastMap)) {
                    self.podcast_last_episode[name] = podcastLastMap[name]
                }
                for (const [name, entry] of Object.entries(self.podcast_last_episode)) {
                    if (podcastLastMap[name] === undefined) {
                        podcastLastMap[name] = JSON.parse(JSON.stringify(entry))
                    }
                }

                // dir_last_episode: per-key merge via dedicated map
                const dirLastMap = _store["dir-last-episode"]
                for (const name of Object.keys(dirLastMap)) {
                    self.dir_last_episode[name] = dirLastMap[name]
                }
                for (const [name, entry] of Object.entries(self.dir_last_episode)) {
                    if (dirLastMap[name] === undefined) {
                        dirLastMap[name] = JSON.parse(JSON.stringify(entry))
                    }
                }

                // Episode times: merge with Math.max (further progress wins)
                for (let i = 0; i < localStorage.length; i++) {
                    const k = localStorage.key(i)
                    if (k.startsWith('time_')) {
                        const url = k.slice(5)
                        const localTime = parseFloat(localStorage.getItem(k))
                        if (!_timesMap.has(url)) {
                            _timesMap.set(url, localTime)
                        }
                    }
                }
                _timesMap.forEach((remoteTime, url) => {
                    const localTime = parseFloat(localStorage.getItem('time_' + url)) || 0
                    const merged = Math.max(localTime, remoteTime)
                    localStorage.setItem('time_' + url, merged)
                    if (merged > remoteTime) {
                        _timesMap.set(url, merged)
                    }
                })

                self._yjsSyncing = false
                self._yjsInitialSyncDone = true
            })
        },

        // --- Media element ---

        _initMedium(el) {
            let cur = localStorage.getItem(this.key('time'))
            if (cur) {
                el.currentTime = cur
            }
            let episode_rate = localStorage.getItem(this.key('rate'))
            if (episode_rate) {
                el.playbackRate = parseFloat(episode_rate)
            } else {
                let podcast_rate = localStorage.getItem(`rate_${this.current.podcast.name}`)
                if (podcast_rate) {
                    el.playbackRate = parseFloat(podcast_rate)
                }
            }
            window.v = el
            if ('mediaSession' in navigator) {
                var artist = ''
                if (this.current.episode.index !== null) {
                    artist += `${this.current.episode.index}`
                }
                if (this.current.episode.number !== null) {
                    artist += `/${this.current.episode.number}`
                }
                if (artist !== '') {
                    artist += ' '
                }
                artist += this.current.podcast.name
                navigator.mediaSession.metadata = new MediaMetadata({
                    title: this.current.episode.name,
                    artist: artist,
                    album: this.current.episode.album || this.current.podcast.name,
                });
                navigator.mediaSession.setActionHandler('nexttrack', () => { emit('gotonext', {playing: true}) });
                navigator.mediaSession.setActionHandler('previoustrack', () => { emit('gotoprev', {playing: true}) });
                navigator.mediaSession.playbackState = 'paused'
            }
        },

        setupMedium(el) {
            const self = this
            // Clean up previous media element's window listeners
            if (_mediaEl && _mediaEl._cleanup) _mediaEl._cleanup()
            _mediaEl = el
            el._lastUrl = self.current.episode?.url
            self._initMedium(el)

            // Window-level event listeners (replaces Alpine .window modifier)
            const handlers = {
                pleaseplay: () => el.play(),
                pleasepause: () => el.pause(),
                pleaseffwd: () => { el.currentTime += 10 },
                pleaserewind: (e) => { el.currentTime -= (e.detail?.time || 10) },
                pleasevolume: (e) => { el.volume = e.detail.volume },
                pleasesetspeed: (e) => { el.playbackRate = e.detail.speed },
                setcurrenttime: (e) => { el.currentTime = e.detail.value },
            }
            for (const [name, handler] of Object.entries(handlers)) {
                window.addEventListener(name, handler)
            }
            el._cleanup = () => {
                for (const [name, handler] of Object.entries(handlers)) {
                    window.removeEventListener(name, handler)
                }
            }

            // Element-level event listeners
            el.addEventListener('play', () => {
                el.volume = 1
                self.current.playing = true
            })
            el.addEventListener('pause', () => {
                self.current.playing = false
                if (el.currentTime === el.duration) {
                    self.onended()
                }
            })
            el.addEventListener('timeupdate', () => {
                localStorage.setItem(self.key('time'), el.currentTime)
                self.current_progress = {cur: el.currentTime, duration: el.duration}
                self._syncEpisodeTime(self.current.episode.url, el.currentTime)
            })
            el.addEventListener('ratechange', () => {
                localStorage.setItem(self.key('rate'), el.playbackRate)
                localStorage.setItem(`rate_${self.current.podcast.name}`, el.playbackRate)
                self.current_speed = el.playbackRate
            })
            el.addEventListener('loadeddata', () => {
                requestAnimationFrame(() => el.scrollIntoView())
            })
            if (self.current.playing) {
                requestAnimationFrame(() => el.play())
            }
            self.current_progress = {cur: el.currentTime, duration: el.duration}
        },

        // Called from v-effect to detect episode changes and reload media
        // (replaces $watch('current.episode', ...) in Alpine medium x-bind)
        async reloadMedium(episodeUrl) {
            if (!_mediaEl || !episodeUrl) return
            if (_mediaEl._lastUrl === episodeUrl) return
            _mediaEl._lastUrl = episodeUrl
            await nextTick() // wait for :src binding to update
            _mediaEl.load()
            this._initMedium(_mediaEl)
            if (this.current.playing) {
                await nextTick()
                _mediaEl.play()
            }
            this.current_progress = {cur: _mediaEl.currentTime, duration: _mediaEl.duration}
        },

        async videoClick(event, el) {
            let xpos = event.offsetX / event.target.clientWidth
            let ypos = event.offsetY / event.target.clientHeight
            if (xpos < 0.25) {
                info('rewind')
                emit('pleaserewind')
            } else if (xpos > 0.75) {
                info('ffwd')
                emit('pleaseffwd')
            } else {
                if (ypos > 0.80) {
                    info('PIP')
                    try {
                        await el.requestPictureInPicture()
                    } catch(e) {
                        info(JSON.stringify(e))
                    }
                } else {
                    info('toggling')
                    if (this.current.playing) {
                        emit('pleasepause')
                    } else {
                        emit('pleaseplay')
                    }
                }
            }
        },
    }
}

// --- Countdown ---

function Countdown() {
    return {
        motionreplaytimer: null,
        countdown: null,
        remaining_time: 0,
        time: loadFromLS('countdown_time', 1),
        _motionListener: null,

        initCountdown() {
            const self = this
            self._motionListener = (e) => {
                let motion = (
                    e.acceleration.x * e.acceleration.x
                    + e.acceleration.y * e.acceleration.y
                    + e.acceleration.z * e.acceleration.z
                )
                if (motion > 400 && self.motionreplaytimer) {
                    clearTimeout(self.motionreplaytimer)
                    self.motionreplaytimer = null
                    info('shaked, again!!')
                    self.start()
                }
            }
            window.addEventListener('devicemotion', self._motionListener)
        },

        start() {
            const self = this
            emit('pleaseplay')
            self.remaining_time = self.time * 60
            if (self.countdown) {
                clearInterval(self.countdown)
                self.countdown = null
            }
            self.countdown = setInterval(() => {
                self.remaining_time -= 1
                if (self.remaining_time == 4) emit('pleasevolume', {volume: 0.8})
                if (self.remaining_time == 3) emit('pleasevolume', {volume: 0.6})
                if (self.remaining_time == 2) emit('pleasevolume', {volume: 0.4})
                if (self.remaining_time == 1) emit('pleasevolume', {volume: 0.2})
                if (self.remaining_time <= 0) {
                    emit('pleasepause')
                    emit('pleaserewind')
                    self.remaining_time = 0
                    clearInterval(self.countdown)
                    self.countdown = null
                    if (self.motionreplaytimer) { clearTimeout(self.motionreplaytimer) }
                    self.motionreplaytimer = setTimeout(() => {
                        info('end of time to shake to play')
                        self.motionreplaytimer = null
                        emit('pleaserewind', {time: 50})
                    }, 10000)
                }
            }, 1000)
        },

        toggleCountdown() {
            if (this.countdown) {
                clearInterval(this.countdown)
                this.countdown = null
                emit('pleasepause')
                emit('pleaserewind')
                this.remaining_time = 0
            } else {
                this.start()
            }
        },
    }
}

// --- Progress ---

function Progress() {
    return {
        disabled: true,
        disabled_timer: null,
        setTimer() {
            if (this.disabled_timer !== null) {
                clearTimeout(this.disabled_timer)
            }
            this.disabled_timer = setTimeout(() => { this.disabled = true; this.disabled_timer = null }, 5000)
        },
        toggleDisabled() {
            this.disabled = !this.disabled
            if (!this.disabled) {
                this.setTimer()
            } else if (this.disabled_timer !== null) {
                clearTimeout(this.disabled_timer)
                this.disabled_timer = null
            }
        },
    }
}

Player

Audio/video player below the listings.

The Alpine x-bind“medium”= spread pattern is replaced by setupMedium($el) called from @vue:mounted, which attaches all event listeners explicitly.

Episode change detection for seamless track transitions uses a v-effect calling reloadMedium(current.episode?.url) (in listings-ui hidden effects). When the URL changes, it waits for the :src binding to update via nextTick, then calls .load() on the media element. This avoids destroying/recreating the element (critical for uninterrupted background playback when screen is locked).

// clean up old caches
self.addEventListener('activate', event => {
    const currentCaches = [CACHE_NAME];
    event.waitUntil(
        caches.keys().then(cacheNames => {
            return cacheNames.filter(cacheName => !currentCaches.includes(cacheName));
        }).then(cachesToDelete => {
            return Promise.all(cachesToDelete.map(cacheToDelete => {
                return caches.delete(cacheToDelete);
            }));
        }).then(() => self.clients.claim())
    );
});

// taken from https://developer.mozilla.org/en-US/docs/Web/Progressive_web_apps/Guides/Caching

async function cacheFirst(request) {
    const cachedResponse = await caches.match(request);
    if (cachedResponse) {
        console.log(`Get ${request.url} from the cache (cacheFirst)`)
        return cachedResponse;
    }
    try {
        const networkResponse = await fetch(request, { mode: 'cors', credentials: 'omit' });
        console.log(`cacheFirst: ${request.url}: ${request.credentials}, ${networkResponse.ok}, ${networkResponse.status}, ${networkResponse.statusText}`)
        if (networkResponse.ok) {
            console.log(`cacheFirst: Put ${request.url} into the cache`)
            const cache = await caches.open(CACHE_NAME);
            cache.put(request, networkResponse.clone());
        }
        else
        {
            console.log(`cacheFirst: bad response with ${request.url}`)
        }
        return networkResponse;
    } catch (error) {
        console.log(`cacheFirst: error with ${request.url}`)
        return Response.error();
    }
}

async function networkFirst(request) {
    try {
        const networkResponse = await fetch(request);
        if (networkResponse.ok) {
            const cache = await caches.open(CACHE_NAME);
            console.log(`Put ${request.url} into the cache (networkfirst), ${request.credentials}`)
            cache.put(request, networkResponse.clone());
        }
        return networkResponse;
    } catch (error) {
        const cachedResponse = await caches.match(request);
        console.log(`Got ${request.url} from the cache (networkfirst)`)
        return cachedResponse || Response.error();
    }
}

self.addEventListener('fetch', event => {
    const dynamicUrlsRegex = new RegExp(`^.*(${DYNAMIC_URLS.join('|')}).*`);
    if (event.request.url.match(dynamicUrlsRegex)) {
        console.log(`Dealing with ${event.request.url} (networkfirst)`)
        event.respondWith(networkFirst(event.request));
    } else if (event.request.url.includes("/ipfs/") || event.request.url.includes("unpkg.com/")) {
        console.log(`Dealing with ${event.request.url} (cacheFirst)`)
        event.respondWith(cacheFirst(event.request));
    }
    else
    {
        console.log(`Leaving ${event.request.url} to the browser`)
    }
});
