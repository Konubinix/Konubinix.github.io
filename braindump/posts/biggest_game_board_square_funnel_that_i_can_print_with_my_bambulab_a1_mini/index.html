Biggest Game Board Square Funnel That I Can Print With My Bambulab A1 Mini

Fleeting

using build123d

from build123d import *

neck_size = 40
top_size = 40
bottom_size = 180

height = 90
neck_heigh = 20

wall_thickness = 1

def pyramid(bottom_size, top_size, height):
    with BuildPart() as p:
        with BuildSketch(Plane.XY) as bottom_sketch:
            Rectangle(bottom_size, bottom_size)

        with BuildSketch(Plane.XY.offset(height)) as top_sketch:
            Rectangle(top_size, top_size)

        loft()

        with BuildSketch(Plane.XY.offset(height)) as top_sketch:
            Rectangle(top_size, top_size)

        with BuildSketch(Plane.XY.offset(height + neck_heigh)) as neck_sketch:
            Rectangle(top_size, top_size)

        loft()

    return p.part

with BuildPart() as funnel:

    add(pyramid(bottom_size, top_size, height))
    add(pyramid(bottom_size - wall_thickness*2,
               top_size - wall_thickness*2,
               height), mode=Mode.SUBTRACT)

export_stl(funnel.part, "/home/sam/test/next/config/config/a.stl")
