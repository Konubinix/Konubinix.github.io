how to create a reproducible emacs.el

(setq pwd (expand-file-name default-directory))
(setq package-user-dir pwd)
(require 'package)
(add-to-list 'package-archives '("melpa" . "https://melpa.org/packages/") t)
(package-initialize)
(package-refresh-contents)

(package-install 'org-roam)
(setq org-roam-directory pwd)
(setq org-roam-db-location (expand-file-name "roam.db" pwd))
