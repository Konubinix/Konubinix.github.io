#!/usr/bin/env python3
# -*- coding:utf-8 -*-
from helpers import ntfy, lowpower, wifi, setup_espnow, espnow, runner, KV
import time

import json
import machine
import neopixel
import network
import esp
import gc
import ntptime
import requests
lowpower()

led = machine.Pin(2, machine.Pin.OUT)
led.off()  # on
time.sleep(2)
led.on()  # off

leds = {
    "left": neopixel.NeoPixel(machine.Pin(4), 5),
    "right": neopixel.NeoPixel(machine.Pin(5), 5),
}

def run_command(name, command):
    r, g, b, l = command
    now = time.localtime()
    h = now[3]
    m = now[4]
    print(f"{h:02d}:{m:02d} - {gc.mem_free()}: {name}: Running command: {command}")
    command = (int(r*l/255),int(g*l/255),int(b*l/255),)
    led = leds[name]
    led.fill(command)
    led.write()

class Connection:
    def __init__(self):
        pass

    def connect(self):
        run_command("left", [0, 0, 0, 0])
        run_command("right", [0, 0, 0, 0])
        wlan, _ = setup_espnow()
        wifi.on()
        ntfy(f"""mac: {wlan.config('mac')}, channel: {wlan.config('channel')}""",
             channel="iotheart")

        run_command("left", [0, 0, 0, 0])
        run_command("right", [0, 50, 0, 50])

        try:
            ntptime.settime()
        except OSError as e:
            ntfy(f"""{e}
while trying to set the time""", channel="iotheart")

kv = KV()
value = kv.follow("heart", wait=2)

def try_get():
    try:
        return next(value)
    except Exception as e:
        print(f"{e}")
        return None

def step():
    print("Do step")
    commands = try_get()
    if commands is None:
        print("Got nothing, using a default value")
        commands = {
            "left": [50, 0, 0, 50],
            "right": [50, 0, 0, 50],
        }
    else:
        commands = json.loads(commands)
    run_command("left", commands["left"])
    run_command("right", commands["right"])

initialsleeptime = 1

sleeptime = initialsleeptime

@runner()
def run():
    connection = Connection()
    connection.connect()

    while True:
        if espnow.any():
            try:
                data = espnow.recv()
                ntfy(f"Received via ESPNow: {data}", channel="iotheart")
                print(f"Got via espnow {data}")
                payload = json.loads(data[1])
                resp = requests.post(f"http://home/iothelper/{payload['path']}", json=payload["json"], timeout=2)
                if resp.status_code != 200:
                    ntfy(f"""{resp.reason}""", channel="iotheart")

            except Exception as e:
                print(f"Error got from ESPNow: {e}")
                ntfy(f"""{e}""", channel="iotheart")
        try:
            step()
            sleeptime = initialsleeptime
        except Exception as e:
            sleeptime *= 2
            if sleeptime > 60:
                sleeptime = 60
            try:
                now = time.localtime()
                h = now[3]
                m = now[4]
                ntfy(f"""{e}\nwill wait for {sleeptime}s""", channel="iotheart")
            except Exception as e2:
                print(f"{h:02d}:{m:02d}: Could not notify: {e2}")

        print(f"Waiting for {sleeptime}s")
        machine.lightsleep(sleeptime * 1000)
