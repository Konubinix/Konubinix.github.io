cat<<EOF > /tmp/main.py
#!/usr/bin/env python3
# -*- coding:utf-8 -*-
import machine
import time
import ntptime
import requests

led = machine.Pin(2, machine.Pin.OUT)
led.off()  # on
time.sleep(2)
led.on()  # off

ntptime.settime()

time.sleep(10) # to have the time to update the program

while True:
    now = time.localtime()
    h = now[3]
    m = now[4]
    try:
        requests.post(
            "http://home:9705/i",
            headers={"title": f"{h:02d}:{m:02d} iot battery"},
            data=f"""Hello""")
    except Exception as e:
        pass # most likely the night, when the wifi is off
    machine.lightsleep(3600 * 1000)

EOF
