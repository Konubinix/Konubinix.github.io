import time

import plyer
from kivy.app import App
from jnius import autoclass
from kivy.clock import Clock
from kivy.lang import Builder
from kivy.uix.boxlayout import BoxLayout
from kivy.core.camera import Camera
from datetime import datetime, timedelta
from kivy.graphics import Color, Rectangle
from kivy.properties import StringProperty
import os
from datetime import datetime
from kivy.uix.screenmanager import Screen, ScreenManagerException, ScreenManager
from helpers.osc import oschandler, to_service
from helpers.wakelock import WakeLock
from logging import getLogger

logger = getLogger(__name__)

wl = WakeLock('pomodorotimer', wakeup=True)

pink = (1, 0.75, 0.8, 1)
cyan = (0, 1, 1, 1)
grey = (0.3, 0.3, 0.3, 1)
green = (0, 0.9, 0.3, 1)
blue = (0, 0.8, 0.8, 1)
yellow = (0.8, 0.8, 0, 1)
colors = {"idle": grey, "pomodoro": pink, "break": green, "interruption": blue, "exterruption": blue,
          "twominutes": yellow}

Builder.load_string("""

<Pomodoro>:
    id: pomodoro
    orientation: "vertical"

    BoxLayout:
        orientation: "horizontal"
        size_hint: (1, 0.5)
        Label:
            id: endtime
            text: ""
            font_size: 100
            color: (0, 0, 0, 1)

        Label:
            id: countdown
            text: ""
            font_size: 100
            color: (0, 0, 0, 1)

    ProgressBar:
        id: progress

    BoxLayout:
        orientation: "horizontal"
        size_hint: (1, 0.5)
        Button:
            disabled: pomodoro.state not in ("idle", "break")
            text: "Start"
            on_press: root.do("start")
            color: (0, 0, 0, 1)
            background_color: {pink}

        Button:
            text: "Break"
            disabled: pomodoro.state != "idle"
            on_press: root.do("short_break")
            color: (0, 0, 0, 1)
            background_color: {green}

        Button:
            disabled: pomodoro.state != "idle"
            text: "Long Break"
            on_press: root.do("long_break")
            color: (0, 0, 0, 1)
            background_color: {green}

        Button:
            disabled: pomodoro.state not in ("pomodoro", "interruption")
            text: "Interruption"
            on_press: root.do("interruption")
            color: (0, 0, 0, 1)
            background_color: {blue}

        Button:
            text: "Exterruption"
            disabled: pomodoro.state not in ("pomodoro", "exterruption")
            on_press: root.do("exterruption")
            color: (0, 0, 0, 1)
            background_color: {blue}

        Button:
            disabled: pomodoro.state not in ("pomodoro", "twominutes")
            text: "Two minutes"
            on_press: root.do("twominutes")
            color: (0, 0, 0, 1)
            background_color: {yellow}


# copy from the source code of kivy, but increased
<ProgressBar>:
    canvas:
        Color:
            rgb: 1, 1, 1
        BorderImage:
            border: (12, 12, 12, 12)
            pos: self.x, self.y
            size: self.width, self.height
            source: 'atlas://data/images/defaulttheme/progressbar_background'
        BorderImage:
            border: [int(min(self.width * (self.value / float(self.max)) if self.max else 0, 12))] * 4
            pos: self.x, self.y
            size: self.width * (self.value / float(self.max)) if self.max else 0, self.height
            source: 'atlas://data/images/defaulttheme/progressbar'

""".format(**{"blue": blue, "yellow": yellow, "pink": pink, "cyan": cyan, "grey": grey, "green": green, "yellow": yellow}))

class PomodoroScreen(Screen):
    def __init__(self, *args, **kwargs):
        super(PomodoroScreen, self).__init__(*args, **kwargs)
        self.pomodorotimer = Pomodoro()
        self.add_widget(self.pomodorotimer)
        self.bind(on_enter=self.enter_handler)
        self.bind(on_pre_leave=self.pre_leave_handler)
        oschandler("pomodoro:state")(self.pomodorotimer.show)

    def pre_leave_handler(self, _):
        self.pomodorotimer.leave()

    def enter_handler(self, _):
        from plyer import orientation
        orientation.set_sensor()
        self.pomodorotimer.enter()

class Pomodoro(BoxLayout):
    state = StringProperty("idle")

    def _update_rect(self, instance, value):
        self.rect.pos = instance.pos
        self.rect.size = instance.size

    def __init__(self, **kwargs):
        super(Pomodoro, self).__init__(**kwargs)
        with self.canvas.before:
            self.color = Color(1, 0, 0, 1)
            self.rect = Rectangle(size=self.size, pos=self.pos)

        self.end = 0
        self.clock = None
        self.bind(size=self._update_rect, pos=self._update_rect)

    def show(self, msg):
        self.state = msg["state"]
        self.end = msg["end"]
        self.color.rgb = colors[self.state][:3]
        if self.end:
            self.ids.progress.max = msg["total"]
            self.ids.endtime.text = datetime.fromtimestamp(self.end).strftime("%H:%M")
        else:
            self.ids.progress.value = 0
        self.update()

    def update(self, *_):
        if self.end:
            remaining = max(0, int(self.end - time.time()))
            self.ids.progress.value = self.ids.progress.max - remaining
            self.ids.countdown.text = "{:02d}:{:02d}".format(remaining // 60, remaining % 60)
        else:
            self.ids.endtime.text = (datetime.now() + timedelta(minutes=25)).strftime("%H:%M")
            self.ids.countdown.text = ""

    def enter(self):
        self.clock = Clock.schedule_interval(self.update, 1)
        wl.acquire()
        to_service("pomodoro:show")

    def leave(self):
        Clock.unschedule(self.clock)
        self.clock = None
        wl.release()

    def do(self, what):
        to_service("pomodoro:do", {"what": what})

class PomodoroApp(App):

    @staticmethod
    def populate(sm):
        try:
            sm.get_screen('pomodorotimer')
            return
        except ScreenManagerException:
            pass # not populated yet

        sm.add_widget(PomodoroScreen(name='pomodorotimer'))

    def build(self):
        sm = ScreenManager()
        self.populate(sm)
        return sm


def run():
    PomodoroApp().run()
